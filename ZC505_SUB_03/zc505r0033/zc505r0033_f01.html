FORM get_data .

  DATA : lcl_pur     TYPE REF TO zc505cl002,
         lv_code,
         lv_msg(100),
         lt_data     TYPE zc505tt005.

  IF lcl_pur IS NOT BOUND.

    CREATE OBJECT lcl_pur.

  ENDIF.

  CALL METHOD lcl_pur->get_order_data
    EXPORTING
      it_plnbez = so_pln[]
      iv_pwerk  = pa_werk
    IMPORTING
      ev_code   = lv_code
      ev_msg    = lv_msg
    CHANGING
      et_data   = lt_data.

  IF lv_code NE 'S'.
    MESSAGE s000 WITH lv_msg.
    LEAVE LIST-PROCESSING.
  ENDIF.

  MOVE-CORRESPONDING lt_data TO gt_data.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING  pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE gs_fcat-fieldname.
    WHEN 'GAMNG'.
      gs_fcat-qfieldname = 'GMEIN'.
    WHEN 'PSMNG'.
      gs_fcat-qfieldname = 'AMEIN'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  IF gcl_handler IS NOT BOUND.

    CREATE OBJECT gcl_handler.

  ENDIF.

  SET HANDLER : gcl_handler->handle_double_click FOR gcl_grid.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_data
      it_fieldcatalog = gt_fcat.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form handle_double_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW
*&      --> E_COLUMN
*&---------------------------------------------------------------------*
FORM handle_double_click  USING    ps_row TYPE lvc_s_row
                                   ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  CLEAR : gs_data.

  READ TABLE gt_data INTO gs_data INDEX ps_row-index.

  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'AUFNR'.
      MESSAGE i000 WITH gs_data-plnbez_maktx.
  ENDCASE.

ENDFORM.
