FUNCTION zc505fm_get_airline_name.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(PI_CARRID) TYPE  S_CARR_ID
*"  EXPORTING
*"     REFERENCE(PE_CARRNAME) TYPE  S_CARRNAME
*"     REFERENCE(PE_CODE) TYPE  CHAR1
*"     REFERENCE(PE_MSG) TYPE  CHAR100
*"----------------------------------------------------------------------

*       Global data declarations

  SELECT SINGLE carrname
    INTO pe_carrname
    FROM scarr
   WHERE carrid EQ pi_carrid.

  IF sy-subrc NE 0.
    pe_code    = 'E'.
    pe_msg     = 'Data not found'.
  ELSE.
    pe_code    = 'S'.
  ENDIF.

ENDFUNCTION.
