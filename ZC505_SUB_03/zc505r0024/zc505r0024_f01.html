FORM get_data .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~matnr a~mtart a~matkl
         b~werks b~pstat
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM mara AS a
    INNER JOIN marc AS b
      ON a~matnr EQ b~matnr
   WHERE a~matnr IN so_matnr.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM get_matnr_desc .

  DATA : lv_code,
         lv_msg(100),
         lv_tabix    TYPE sy-tabix.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    call function 'ZC505FM_GET_MAKTX'
      EXPORTING " program - >funtion / 오른쪽에서 왼쪽으로 데이터가 들어감.
        iv_matnr = gs_data-matnr
        iv_spras = sy-langu
      IMPORTING "Funtion -> Program / 왼쪽에서 오른쪽으로 데이터를 받아온다.
        ev_maktx = gs_data-maktx
        pe_code  = lv_code
        pe_msg   = lv_msg.

*      EXCEPTIONS
*        matnr_required = 1
*        data_not_found = 2
*        OTHERS         = 3.

*    CASE sy-subrc.
*      WHEN 0.
*        MODIFY gt_data FROM gs_data INDEX lv_tabix
*        TRANSPORTING maktx.
*    ENDCASE.

    CASE lv_code.
      WHEN 'S'.
        MODIFY gt_data FROM gs_data INDEX lv_tabix
        TRANSPORTING maktx.
    ENDCASE.

  ENDLOOP.

ENDFORM.

FORM set_fcat_layout .

  gs_layout-zebra = 'X'.
  gs_layout-cwidth_opt = 'A'.
  gs_layout-sel_mode   = 'D'.

  gs_stable-row = 'X'.

     ' '  'MTART' ' '  'MARA'  ' ',
     ' '  'MATKL' ' '  'MARA'  ' ',

FORM make_fcat  USING pv_key
                      pv_field
                      pv_text
                      pv_ref_table
                      pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.
