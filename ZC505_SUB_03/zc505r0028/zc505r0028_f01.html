FORM get_data .

  DATA : lcl_prod    TYPE REF TO zc505cl002,
         lv_code,
         lv_msg(100),
         lt_prod     TYPE zc505tt005.

  CLEAR   : lv_code, lv_msg.
  REFRESH : lt_prod.

  IF lcl_prod IS NOT BOUND.
    CREATE OBJECT : lcl_prod.
  ENDIF.

  CALL METHOD lcl_prod->get_product_order
    EXPORTING
      iv_plnbez = pa_plnb
      iv_pwerk  = pa_pwerk
    IMPORTING
      ev_code   = lv_code
      ev_msg    = lv_msg
      et_data   = lt_prod.

  IF lv_code NE 'S'.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ELSE.
    MOVE-CORRESPONDING lt_prod TO gt_data.
  ENDIF.

ENDFORM.

FORM make_fcat  USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'STATUS'.
      gs_fcat-icon = 'X'.
    WHEN 'AMEIN'.
      gs_fcat-emphasize = 'C300'.
    WHEN 'POSNR'.
      gs_fcat-just = 'C'. " 가운데 정렬
    WHEN 'GAMNG'.
      gs_fcat-qfieldname = 'GMEIN'.
    WHEN 'PSMNG'.
      gs_fcat-emphasize  = 'C300'.
      gs_fcat-qfieldname = 'AMEIN'.
  ENDCASE.

  APPEND gs_fcat TO  gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM set_color .

  DATA : lv_tabix TYPE sy-tabix,
         ls_color TYPE lvc_s_scol.

  CLEAR : lv_tabix, ls_color.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    gs_data-status = icon_alarm.

    IF gs_data-psmng LE 10000.

      ls_color-fname = 'PSMNG'.
      ls_color-color-col = '6'.
      ls_color-color-int = '1'.
      ls_color-color-inv = '0'.

      APPEND ls_color TO gs_data-color.
      CLEAR  ls_color.

    ENDIF.

    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING status color.

  ENDLOOP.

  CLEAR : gs_data.

ENDFORM.
