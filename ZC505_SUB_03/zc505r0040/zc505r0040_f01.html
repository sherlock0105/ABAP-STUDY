FORM get_scarr .

  _clear : gs_scarr gt_scarr.

  SELECT a~carrid a~carrname "scarr
         b~connid b~airpfrom b~airpto b~distance b~distid
    INTO CORRESPONDING FIELDS OF TABLE gt_scarr
    FROM scarr AS a
   INNER JOIN spfli AS b
      ON a~carrid EQ b~carrid
   WHERE a~carrid EQ pa_carr
     AND b~connid IN so_conn.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

*&      --> P_

FORM make_fcat  USING  pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'CARRID'.
      gs_fcat-hotspot = 'X'.
    WHEN 'DISTANCE'.
      gs_fcat-qfieldname = 'DISTID'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_screen .

    EXPORTING

      extension = 5000.

    EXPORTING

      i_default       = 'X'

*&---------------------------------------------------------------------*
*& Form hotspot_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW_ID
*&      --> E_COLUMN_ID
*&---------------------------------------------------------------------*
FORM hotspot_click  USING    ps_row TYPE lvc_s_row
                             ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_scarr INTO gs_scarr INDEX ps_row-index.

  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

  PERFORM get_sflight_info USING gs_scarr-carrid
                                 gs_scarr-connid.

  IF gt_sflight IS INITIAL.
    MESSAGE s000 WITH 'Flight data not found'.
    EXIT.
  ENDIF.

  CALL SCREEN '0101' STARTING AT 20 3.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_sflight_info
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM get_sflight_info USING pv_carrid TYPE sflight-carrid
                            pv_connid TYPE sflight-connid.

  _clear gs_sflight gt_sflight.

  SELECT carrid connid fldate price
         paymentsum currency
    INTO CORRESPONDING FIELDS OF TABLE gt_sflight
    FROM sflight
   WHERE carrid EQ pv_carrid
     AND connid EQ pv_connid.

ENDFORM.
