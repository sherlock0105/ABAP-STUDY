FORM get_data .

  CASE 'X'.
    WHEN rb_func.
      PERFORM get_data_function.

    WHEN rb_meth.
      PERFORM get_data_method.

    WHEN rb_form.
      PERFORM get_data_form.

  ENDCASE.

ENDFORM.

FORM get_data_function .

  DATA : lv_code,
         lv_msg(100),
         lt_data        TYPE zc505tt006.

  call function 'ZC505FM_GET_EKKO'
    EXPORTING
      iv_bukrs = pa_bukrs
    IMPORTING
      ev_code  = lv_code
      ev_msg   = lv_msg
    TABLES
      et_data  = lt_data
      it_ebeln = so_ebeln.

  IF lv_code NE 'S'.
    MESSAGE s000 WITH lv_msg.
    LEAVE LIST-PROCESSING.

  ELSE.
    MESSAGE s000 WITH 'Function 화면입니다.'.
    MOVE-CORRESPONDING lt_data TO gt_data.
  ENDIF.

ENDFORM.

FORM get_data_method .

         lv_msg(100),

    EXPORTING

  ELSE.
    MESSAGE s000 WITH 'Method 화면입니다.'.

  ENDIF.

FORM get_data_form .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~ebeln a~bukrs a~bstyp a~bsart
         b~ebelp b~matnr b~werks b~ktmng b~meins
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM ekko AS a
   INNER JOIN ekpo AS b
      ON a~ebeln EQ b~ebeln
   WHERE a~bukrs EQ pa_bukrs
     AND a~ebeln IN so_ebeln.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ELSE.
    MESSAGE s000 WITH 'Form 화면입니다.'.
  ENDIF.

ENDFORM.

FORM set_color .

  DATA : ls_color TYPE lvc_s_scol,
         lv_tabix TYPE sy-tabix.

  CLEAR  : ls_color, lv_tabix.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    IF gs_data-ktmng LE 5000.

      ls_color-fname     = 'KTMNG'.
      ls_color-color-col = '3'.
      ls_color-color-int = '0'.
      ls_color-color-inv = '0'.

      APPEND ls_color TO gs_data-color.
      CLEAR  ls_color.

    ENDIF.

    IF gs_data-meins EQ 'EA'.

      gs_data-status = icon_transport.

    ELSE.

      gs_data-status = icon_flight.

    ENDIF.

    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING status color.

  ENDLOOP.

ENDFORM.
