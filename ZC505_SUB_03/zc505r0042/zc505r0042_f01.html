FORM get_data .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~blart a~budat "BKPF
         b~belnr b~buzei b~bschl b~koart b~gsber "BSEG
         b~shkzg b~hkont b~dmbtr b~pswsl
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM bkpf AS a
   INNER JOIN bseg AS b
      ON a~bukrs EQ b~bukrs
     AND a~belnr EQ b~belnr
     AND a~belnr EQ b~belnr
   WHERE a~bukrs EQ pa_bukrs
     AND a~gjahr EQ pa_gjahr.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM make_fcat  USING  pv_key
                       pv_field
                       pv_text
                       pv_ref_table
                       pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'BELNR'.
      gs_fcat-hotspot = 'X'.
    WHEN 'DMBTR'.
      gs_fcat-cfieldname = 'PSWSL'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  IF gcl_handler IS NOT BOUND.
    CREATE OBJECT gcl_handler.
  ENDIF.

  SET HANDLER : gcl_handler->handle_hotspot_click FOR gcl_grid,
                gcl_handler->handle_double_click  FOR gcl_grid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_data
      it_fieldcatalog = gt_fcat.

ENDFORM.

                                  ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL. "소계/합계 행 더블클릭 방지
    EXIT.
  ENDIF.

  READ TABLE gt_data INTO gs_data INDEX ps_row-index. ""실제 클릭된 행 데이터 가져오기

  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname. "'BELNR' 컬럼을 클릭했을때만 동작
