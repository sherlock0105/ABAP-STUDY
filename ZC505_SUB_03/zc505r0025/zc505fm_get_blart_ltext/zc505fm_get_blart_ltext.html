FUNCTION zc505fm_get_blart_ltext.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_BLART) TYPE  BLART
*"     REFERENCE(IV_SPRAS) TYPE  SPRAS OPTIONAL
*"  EXPORTING
*"     REFERENCE(EV_LTEXT) TYPE  LTEXT_003T
*"     REFERENCE(EV_CODE) TYPE  CHAR1
*"     REFERENCE(EV_MSG) TYPE  CHAR100
*"----------------------------------------------------------------------

*       Global data declarations

  DATA : lv_spras TYPE sy-langu.

  IF iv_blart IS INITIAL. " 필수 입력값 체크
    ev_code = 'E'.
    ev_msg  = 'Please wirte blart'.
    EXIT.
  ENDIF.

  IF iv_spras IS INITIAL.
    lv_spras = sy-langu.
  ELSE.
    lv_spras = iv_spras.
  ENDIF.

  SELECT SINGLE ltext
    INTO ev_ltext
    FROM t003t
   WHERE blart EQ iv_blart
     AND spras EQ lv_spras.

  IF sy-subrc NE 0.
    ev_code = 'E'.
    ev_msg  = 'Data not found'.
  ELSE.
    ev_code = 'S'.
  ENDIF.

ENDFUNCTION.
