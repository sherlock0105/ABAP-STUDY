FUNCTION zc505fm_get_gsber_gtext.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_GSBER) TYPE  GSBER
*"     REFERENCE(IV_SPRAS) TYPE  SPRAS OPTIONAL
*"  EXPORTING
*"     REFERENCE(EV_GTEXT) TYPE  GTEXT
*"     REFERENCE(EV_CODE) TYPE  CHAR1
*"     REFERENCE(EV_MSG) TYPE  CHAR100
*"----------------------------------------------------------------------

*       Global data declarations

  DATA : lv_spras TYPE sy-langu.

  IF iv_gsber IS INITIAL.
    ev_code = 'E'.
    ev_msg  = 'Please wirte gsber'.
    EXIT.
  ENDIF.

  IF iv_spras IS INITIAL.
    lv_spras = sy-langu.
  ELSE.
    lv_spras = iv_spras.
  ENDIF.

  SELECT SINGLE gtext
    INTO ev_gtext
    FROM tgsbt
   WHERE gsber EQ iv_gsber
     AND spras EQ lv_spras.

  IF sy-subrc NE 0.
    ev_code = 'E'.
    ev_msg  = 'Data not found'.
  ELSE.
    ev_code = 'S'.
  ENDIF.

ENDFUNCTION.
