FORM get_text .

  DATA : lv_code,
         lv_msg(100),
         lv_tabix    TYPE sy-tabix.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.
    CLEAR: lv_code, lv_msg.
    call function 'ZC505FM_GET_BLART_LTEXT'
      EXPORTING
        iv_blart = gs_data-blart
        iv_spras = sy-langu
      IMPORTING
        ev_ltext = gs_data-ltext
        ev_code  = lv_code
        ev_msg   = lv_msg.

    CASE lv_code.
      WHEN 'S'.
        MODIFY gt_data FROM gs_data INDEX lv_tabix
        TRANSPORTING ltext.
    ENDCASE.

    call function 'ZC505FM_GET_GSBER_GTEXT'
      EXPORTING
        iv_gsber = gs_data-gsber
        iv_spras = sy-langu
      IMPORTING
        ev_gtext = gs_data-gtext
        ev_code  = lv_code
        ev_msg   = lv_msg.

    CASE lv_code.
      WHEN 'S'.
        MODIFY gt_data FROM gs_data INDEX lv_tabix
        TRANSPORTING gtext.
    ENDCASE.

  ENDLOOP.

ENDFORM.

*& Form get_posting
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> LT_DATA
*&      <-- LV_CODE
*&      <-- LV_MSG
*&---------------------------------------------------------------------*
FORM get_posting  TABLES   pt_data LIKE gt_data
                  CHANGING pv_code
                           pv_msg.

  SELECT a~blart a~budat
         b~belnr b~buzei b~bschl b~koart
         b~gsber b~shkzg b~dmbtr b~pswsl
    INTO CORRESPONDING FIELDS OF TABLE pt_data
    FROM bkpf AS a
   INNER JOIN bseg AS b
      ON a~bukrs EQ b~bukrs
     AND a~belnr EQ b~belnr
     AND a~gjahr EQ b~gjahr
   WHERE a~bukrs EQ pa_bukrs
     AND a~gjahr EQ pa_gjahr.

  IF sy-subrc NE 0.
    pv_code = 'E'.
    pv_msg  = 'Data not found'.
  ELSE.
    pv_code = 'S'.
  ENDIF.

ENDFORM.

         ls_color TYPE lvc_s_scol.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

*    gs_data-status = icon_transport.

    CASE gs_data-budat.
      WHEN '20101204'.

        ls_color-color-col = 3.
        ls_color-color-int = 0.
        ls_color-color-inv = 0.
        APPEND ls_color TO gs_data-color.
    ENDCASE.

    CASE gs_data-pswsl.

      WHEN 'KRW'.
        gs_data-pswsl  = '원화'.

        ls_color-color-int = 0.
        ls_color-color-inv = 0.
        APPEND ls_color TO gs_data-color.

    ENDCASE.


    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING pswsl color.

  ENDLOOP.
