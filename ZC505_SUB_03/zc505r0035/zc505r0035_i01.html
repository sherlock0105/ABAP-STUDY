MODULE exit_0100 INPUT.

  CALL METHOD : gcl_grid->free, gcl_container->free.

  FREE : gcl_grid, gcl_container.

  CASE gv_okcode.
    WHEN 'CANC'.
      LEAVE PROGRAM.
    WHEN 'BACK' OR 'EXIT'.
      LEAVE TO SCREEN 0.
  ENDCASE.

ENDMODULE.

MODULE user_command_0100 INPUT.

  CASE gv_okcode.
    WHEN 'REFRESH'.
      PERFORM get_data.
      PERFORM refresh_grid.
  ENDCASE.

ENDMODULE.
