FUNCTION zc505fm_get_ekko.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_BUKRS) TYPE  EKKO-BUKRS
*"  EXPORTING
*"     REFERENCE(EV_CODE) TYPE  CHAR1
*"     REFERENCE(EV_MSG) TYPE  CHAR100
*"  TABLES
*"      ET_DATA STRUCTURE  ZC505S0014
*"      IT_EBELN OPTIONAL
*"----------------------------------------------------------------------

*       Global data declarations

  SELECT a~ebeln a~bukrs a~bstyp a~bsart
         b~ebelp b~matnr b~werks b~ktmng b~meins
    INTO CORRESPONDING FIELDS OF TABLE et_data
    FROM ekko AS a
   INNER JOIN ekpo AS b
      ON a~ebeln EQ b~ebeln
   WHERE a~bukrs EQ iv_bukrs.

  IF sy-subrc NE 0.
    ev_code = 'E'.
    ev_msg  = 'Data not found'.
  ELSE.
    ev_code = 'S'.
  ENDIF.

ENDFUNCTION.
