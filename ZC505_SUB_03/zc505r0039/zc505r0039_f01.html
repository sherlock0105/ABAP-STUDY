FORM make_fcat  USING  pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.

  CASE pv_field.
    WHEN 'DISTANCE'.
      gs_fcat-qfieldname = 'DISTID'.
    WHEN 'FLTYPE'.
      gs_fcat-checkbox = 'X'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM get_scarr USING pv_carrid TYPE scarr-carrid.

FORM display_pop .

  CREATE OBJECT gcl_container_POP
    EXPORTING
      container_name = 'GCL_CONTAINER_POP'.

  CREATE OBJECT gcl_grid_pop
    EXPORTING
      i_parent = gcl_container_pop.

  gs_layout-no_toolbar = 'X'.

  CALL METHOD gcl_grid_pop->set_table_for_first_display
    EXPORTING
      is_layout       = gs_layout
    CHANGING
      it_fieldcatalog = gt_fcat_pop
      it_outtab       = gt_scarr.

ENDFORM.
