FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  IF gcl_handler IS NOT BOUND.

    CREATE OBJECT gcl_handler.

  ENDIF.

  SET HANDLER : gcl_handler->handle_double_click FOR gcl_grid.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_scarr
      it_fieldcatalog = gt_fcat.

ENDFORM.

* 더블클릭이 발생한 행의 carrid 값을 이용해서 spfil 의 connid max 값을
* message 로 띄워보자

* 더블클릭이 발생한 행의 carrid 값은? gs_scarr-carrid 에 있다.

  SELECT MAX( connid )
    INTO lv_max_connid
    FROM spfli
   WHERE carrid EQ gs_scarr-carrid.

  IF lv_max_connid IS INITIAL.
    MESSAGE s001.
    EXIT.
  ENDIF.

  MESSAGE i000 WITH lv_max_connid.
