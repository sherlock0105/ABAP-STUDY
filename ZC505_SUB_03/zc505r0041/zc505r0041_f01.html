*& <--  p2        text

FORM get_data .

  CLEAR   : gs_mat.
  REFRESH : gt_mat.

  SELECT a~matnr a~vpsta a~mbrsh a~meins
         b~werks b~ekgrp b~dismm b~beskz
         c~maktx "makt
         d~eknam
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM mara AS a
   INNER JOIN marc AS b
      ON a~matnr EQ b~matnr
    LEFT OUTER JOIN makt AS c
      ON a~matnr EQ c~matnr
     AND c~spras EQ sy-langu
    LEFT OUTER JOIN t024 AS d
      ON b~ekgrp EQ d~ekgrp
   WHERE b~werks EQ pa_werks.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM make_fcat  USING  pv_key
                       pv_field
                       pv_text
                       pv_ref_table
                       pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'EKGRP'.
      gs_fcat-emphasize = 'C300'.
    WHEN 'WERKS'.
      gs_fcat-hotspot = 'X'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.


  SET HANDLER : lcl_event_handler=>handle_double_click  FOR gcl_grid,
                lcl_event_handler=>handle_hotspot_click FOR gcl_grid.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_mat
      it_fieldcatalog = gt_fcat.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form handle_double_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW
*&      --> E_COLUMN
*&---------------------------------------------------------------------*
FORM handle_double_click  USING    ps_row TYPE lvc_s_row
                                   ps_col TYPE lvc_s_col.
  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_mat INTO gs_mat INDEX ps_row-index.

  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

  PERFORM get_matnr USING gs_mat-matnr.

  IF gt_mat IS INITIAL.
    MESSAGE s000 WITH 'Matarial data not found'.
    EXIT.
  ENDIF.

  CASE ps_col-fieldname.
    WHEN 'MATNR'.
      CALL SCREEN '0101' STARTING AT 20 3.
  ENDCASE.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_matnr
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_MAT_MATNR
*&---------------------------------------------------------------------*
FORM get_matnr  USING  pv_matnr TYPE mara-matnr.

  CLEAR   : gs_matnr.
  REFRESH : gt_matnr.

  SELECT a~matnr a~mtart a~matkl
         b~maktx "makt
         c~mtbez "t134t
         d~wgbez "t023t
    INTO CORRESPONDING FIELDS OF TABLE gt_matnr
    FROM mara AS a
    LEFT OUTER JOIN makt AS b
      ON a~matnr EQ b~matnr
     AND b~spras EQ sy-langu
    LEFT OUTER JOIN t134t AS c
      ON a~mtart EQ c~mtart
     AND c~spras EQ sy-langu
    LEFT OUTER JOIN t023t AS d
      ON a~matkl EQ d~matkl
     AND d~spras EQ sy-langu
   WHERE a~matnr EQ pv_matnr.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form get_werks
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_MAT_WERKS
*&---------------------------------------------------------------------*
FORM get_werks  USING    pv_matnr TYPE marc-matnr
                         pv_werks TYPE marc-werks.

  CLEAR   : gs_werks.
  REFRESH : gt_werks.

  SELECT a~matnr a~werks a~dispo a~disls
         b~maktx
         c~dsnam
    INTO CORRESPONDING FIELDS OF TABLE gt_werks
    FROM marc AS a
    LEFT OUTER JOIN makt AS b
      ON a~matnr EQ b~matnr
     AND b~spras EQ sy-langu
    LEFT OUTER JOIN t024d AS c
      ON a~werks EQ c~werks
     AND a~dispo EQ c~dispo
   WHERE a~matnr EQ pv_matnr
     AND a~werks EQ pv_werks.

ENDFORM.
