FORM get_data .

  _clear gs_data gt_data.

  DATA : lr_bukrs TYPE RANGE OF bkpf-bukrs,
         lr_gjahr TYPE RANGE OF bkpf-gjahr,

         ls_bukrs LIKE LINE OF lr_bukrs,
         ls_gjahr LIKE LINE OF lr_gjahr.

  IF pa_bukrs IS NOT INITIAL.

    ls_bukrs-low    = pa_bukrs.
    ls_bukrs-sign   = 'I'.
    ls_bukrs-option = 'EQ'.

    APPEND ls_bukrs TO lr_bukrs.
    CLEAR  ls_bukrs.

  ENDIF.

  IF pa_gjahr IS NOT INITIAL.

    ls_gjahr-low    = pa_gjahr.
    ls_gjahr-sign   = 'I'.
    ls_gjahr-option = 'EQ'.

    APPEND ls_gjahr TO lr_gjahr.
    CLEAR  ls_gjahr.

  ENDIF.

  SELECT bukrs gjahr belnr budat
         blart bktxt xrueb
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM bkpf
   WHERE bukrs IN lr_bukrs
     AND gjahr IN lr_gjahr.

  IF sy-subrc NE 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM display_split .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left   " Side to Which Control is Docked
      extension = 5000.                         " Control Extension

  CREATE OBJECT gcl_splitter
    EXPORTING
      parent  = gcl_container                   " Parent Container
      rows    = 2
      columns = 1.

*1행 1열  컨테이너  .
  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1                 " Row
      column    = 1                " Column
    RECEIVING
      Container = gcl_container_data.

*2행 1열  컨테이너  .
  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 2                 " Row
      column    = 1                " Column
    RECEIVING
      Container = gcl_container_acc.
*1행 1열  grid   .
  CREATE OBJECT gcl_grid_data
    EXPORTING
      i_parent = gcl_container_data.                 " Parent Container

*1행 1열  grid  .
  CREATE OBJECT gcl_grid_acc
    EXPORTING
      i_parent = gcl_container_acc.                 " Parent Container

  CALL METHOD gcl_splitter->set_row_height
    EXPORTING
      id     = 1                " Row ID
      height = 40.              " Height

  gs_variant-report = sy-repid.
  gs_variant-handle = 'AA'. "구분자

  SET HANDLER : lcl_event_handler=>handle_hotspot_click FOR gcl_grid_data.

  CALL METHOD gcl_grid_data->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant            " Layout
      i_save          = 'A'                   " Save Layout
      i_default       = 'X'                   " Default Display Variant
      is_layout       = gs_layout             " Layout
    CHANGING
      it_outtab       = gt_data              " Output Table
      it_fieldcatalog = gt_fcat_data.          " Field Catalog

  gs_variant-handle = 'BB'. "구분자
  gs_layout-ctab_fname ='COLOR'. "COLOR 은 아래 acc grid 에만 적용

  CALL METHOD gcl_grid_acc->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant            " Layout
      i_save          = 'A'                   " Save Layout
      i_default       = 'X'                   " Default Display Variant
      is_layout       = gs_layout             " Layout
    CHANGING
      it_outtab       = gt_acc              " Output Table
      it_fieldcatalog = gt_fcat_acc.          " Field Catalog

ENDFORM.

FORM handle_hotspot_click  USING    ps_row TYPE lvc_s_row
                                    ps_col TYPE lvc_s_col.

  REFRESH gt_acc. "클릭했을때 아래 정보 클리어

  CHECK ps_row-rowtype IS INITIAL.

  READ TABLE gt_data INTO gs_data INDEX ps_row-index.

  CHECK sy-subrc EQ 0.

  PERFORM get_acc USING gs_data-bukrs
                        gs_data-gjahr
                        gs_data-belnr.

  IF gt_acc IS INITIAL.
    MESSAGE s000 WITH gs_data-belnr 'DATA IS NOT FOUND' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  PERFORM set_color.

ENDFORM.

FORM set_color .

  DATA : ls_color TYPE lvc_s_scol,
         lv_tabix TYPE sy-tabix,
         ls_acc   LIKE LINE OF gt_acc.

  LOOP AT gt_acc INTO ls_acc.

    lv_tabix = sy-tabix.

    CLEAR ls_color.
    REFRESH ls_acc-color.

    CASE ls_acc-pswsl.

      WHEN 'KRW'.

        ls_color-fname = 'PSWSL'.
        ls_color-color-col = 5.
        ls_color-color-int = 1.
        ls_color-color-inv = 0.

      WHEN 'EUR'.

        ls_color-color-col = 6.
        ls_color-color-int = 1.
        ls_color-color-inv = 0.

    ENDCASE.

    APPEND ls_color TO ls_acc-color.
    CLEAR  ls_color.

    MODIFY gt_acc FROM ls_acc INDEX lv_tabix
    TRANSPORTING color.

  ENDLOOP.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_acc
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_DATA_BUKRS
*&      --> GS_DATA_GJAHR
*&      --> GS_DATA_BELNR
*&---------------------------------------------------------------------*
FORM get_acc  USING    pv_bukrs TYPE bkpf-bukrs
                       pv_gjahr TYPE bkpf-gjahr
                       pv_belnr TYPE bkpf-belnr.

  _clear gs_acc gt_acc.

  SELECT a~belnr a~bldat
         b~buzei b~bschl b~koart b~shkzg b~dmbtr
         b~pswsl b~sgtxt b~kokrs
    INTO CORRESPONDING FIELDS OF TABLE gt_acc
    FROM bkpf AS a
   INNER JOIN bseg AS b
      ON a~belnr EQ b~belnr
   WHERE a~bukrs EQ pv_bukrs
     AND a~gjahr EQ pv_gjahr
     AND a~belnr EQ pv_belnr.

ENDFORM.
