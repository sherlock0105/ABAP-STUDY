FORM get_data .

  _clear gs_cost gt_cost.
  REFRESH gt_cost_del.

  SELECT  a~kokrs   a~kostl  a~bukrs  a~kosar
          a~verak  a~wrbtr  a~waers  a~bkzer
          a~erdat  a~erzet  a~ernam  a~aedat
          a~aezet  a~aenam
          b~ltext
    INTO CORRESPONDING FIELDS OF TABLE gt_cost
    FROM zc522t0033 AS a
    LEFT OUTER JOIN cskt AS b
      ON a~kokrs EQ b~kokrs
     AND a~kostl EQ b~kostl
     AND b~spras EQ sy-langu
   WHERE a~kokrs EQ pa_kokrs
   ORDER BY a~kokrs a~kostl.

ENDFORM.

FORM handle_user_command  USING  pv_ucomm TYPE sy-ucomm.

  CASE pv_ucomm.

    WHEN 'REFRESH'.
      CLEAR gv_okcode.

      PERFORM get_data.
      PERFORM set_style.
      PERFORM refresh_data.

      CHECK gv_screen = '0101'. "0101 이 아니면 REFRESH 까지만 그린다.

    WHEN 'SAVE'.
      CLEAR gv_okcode.

      PERFORM save_data.
      PERFORM get_data.
      PERFORM set_style.
      PERFORM refresh_data.

    WHEN 'CREATE'.
      CLEAR gv_okcode.

      PERFORM create_data.
      PERFORM refresh_data.

    WHEN 'DELETE'.
      CLEAR gv_okcode.

      PERFORM delete_data.
    WHEN 'TOGGLE'.
      CLEAR gv_okcode.

      PERFORM toggle_display.

  ENDCASE.

ENDFORM.

FORM handle_changed_finished  USING    pv_modified
                                       pt_good_cells TYPE lvc_t_modi.
*CSKT

  DATA : ls_modi TYPE lvc_s_modi.

  CHECK pv_modified IS NOT INITIAL.

  LOOP AT pt_good_cells INTO ls_modi.

    READ TABLE gt_cost INTO gs_cost INDEX ls_modi-row_id.

    IF sy-subrc NE 0.
      CONTINUE.
    ENDIF.

    gs_cost-mark = 'X'.

    CASE ls_modi-fieldname.
      WHEN 'KOSTL'.
        SELECT SINGLE ltext
          INTO gs_cost-ltext
          FROM cskt
          WHERE spras EQ sy-langu
          AND kokrs EQ pa_kokrs
          AND kostl EQ gs_cost-kostl.

    ENDCASE.

    MODIFY gt_cost FROM gs_cost INDEX ls_modi-row_id
    TRANSPORTING mark ltext.

  ENDLOOP.

  PERFORM refresh_data.

ENDFORM.

FORM save_data .

  DATA: ls_save   TYPE zc522t0033,
        lt_save   LIKE TABLE OF ls_save,
        lt_del    LIKE TABLE OF ls_save,
        lv_commit.

  CLEAR ls_save.

  CALL METHOD gcl_grid_1->check_changed_data.

  IF gt_row IS NOT INITIAL.

    MOVE-CORRESPONDING gt_cost_del TO lt_del.
    DELETE zc522t0033 FROM TABLE lt_del.
    MESSAGE s003.

    IF sy-dbcnt > 0.

      lv_commit = 'X'.

    ENDIF.

  ELSE.

  ENDIF.

  LOOP AT gt_cost INTO gs_cost WHERE mark EQ 'X'.

    MOVE-CORRESPONDING gs_cost TO ls_save.

    CHECK gs_cost-kostl IS NOT INITIAL. "KEY 값 없으면 저장 막음

    IF ls_save-erdat IS INITIAL. "erdat 생성 날짜

      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.

    ELSE. "기존에 있었을 경우
      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.

    MODIFY zc522t0033 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.
    COMMIT WORK AND WAIT.

    REFRESH gt_cost_del.

    PERFORM clear_mark.
    MESSAGE s002.

  ELSE.

    MESSAGE s000 WITH 'Data was not changed' DISPLAY LIKE 'E'.

  ENDIF.


ENDFORM.
