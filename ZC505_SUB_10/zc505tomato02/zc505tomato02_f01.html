FORM handle_changed_finished  USING    pv_modified
                                       pt_good_cells TYPE lvc_t_modi.

  DATA : ls_modi TYPE lvc_s_modi.

  CHECK pv_modified IS NOT INITIAL.


  LOOP AT pt_good_cells INTO ls_modi.

    READ TABLE gt_data INTO gs_data INDEX ls_modi-row_id.

    IF sy-subrc NE 0.
      CONTINUE.
    ENDIF.

    gs_data-mark = 'X'.

    CASE ls_modi-fieldname.
      WHEN 'MEALNUMBER'.

        SELECT SINGLE mealtype
          INTO gs_data-mealtype
          FROM zmeal
         WHERE carrid EQ pa_carr
          AND  mealnumber EQ gs_data-mealnumber.

        SELECT SINGLE text
          INTO gs_data-text
          FROM zmealt
         WHERE carrid EQ pa_carr
           AND mealnumber EQ gs_data-mealnumber
           AND sprache EQ sy-langu.


    ENDCASE.

    MODIFY gt_data FROM gs_data INDEX ls_modi-row_id
    TRANSPORTING mark mealtype text.
  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH 'First, select rows'.
    EXIT.
  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_data INTO gs_data INDEX gs_row-index.

    IF sy-subrc EQ 0.

      APPEND gs_data TO gt_data_del.
      CLEAR  gs_data.

      DELETE gt_data INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM get_data_form .

  _clear gs_data gt_data.
  REFRESH gt_data_del.

*  SELECT a~carrid a~mealnumber a~mlcnt a~meins
*         a~erdat a~erzet a~ernam
*         a~aedat a~aezet a~aenam
*         b~mealtype
*         c~text
*    INTO CORRESPONDING FIELDS OF TABLE gt_data
*    FROM zc505t0045 AS a
*    LEFT OUTER JOIN zmeal AS b
*      ON a~carrid EQ b~carrid
*     AND a~mealnumber EQ b~mealnumber
*    LEFT OUTER JOIN zmealt AS c
*      ON a~carrid EQ c~carrid
*     AND a~mealnumber EQ c~mealnumber
*     AND c~sprache EQ sy-langu
*   WHERE a~carrid EQ pa_carr
*   ORDER BY a~carrid a~mealnumber.

  SELECT a~carrid, a~mealnumber, a~mlcnt,
         a~meins,  a~erdat,         a~erzet,
         a~ernam, a~aedat,         a~aezet,
         a~aenam, b~mealtype,         c~text,
         CASE
           WHEN a~mlcnt = 0 THEN @icon_red_light
           WHEN a~mlcnt < 10 THEN @icon_yellow_light
           ELSE @icon_green_light
         END AS icon
    INTO CORRESPONDING FIELDS OF TABLE @gt_data
    FROM zc505t0045 AS a
    LEFT OUTER JOIN zmeal  AS b
      ON a~carrid     = b~carrid
     AND a~mealnumber = b~mealnumber
    LEFT OUTER JOIN zmealt AS c
      ON a~carrid     = c~carrid
     AND a~mealnumber = c~mealnumber
     AND c~sprache    = @sy-langu
   WHERE a~carrid     = @pa_carr
   ORDER BY a~carrid, a~mealnumber.

ENDFORM.
