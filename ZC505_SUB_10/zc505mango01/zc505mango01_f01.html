FORM get_data .

  DATA : lr_carrid TYPE RANGE OF scarr-carrid,
         lr_fldate TYPE RANGE OF sbook-fldate.

  IF pa_car IS NOT INITIAL.

    lr_carrid = VALUE #( BASE lr_carrid
                      ( low    = pa_car
                        sign   = 'I'
                        option = 'EQ' )
                       ).

  ENDIF.

  IF pa_fld IS NOT INITIAL.

    lr_fldate = VALUE #( BASE lr_fldate
                      ( low    = pa_fld
                        sign   = 'I'
                        option = 'EQ' )
                       ).

  ENDIF.

  SELECT *
    FROM zc505cds0025
    INTO CORRESPONDING FIELDS OF TABLE @gt_data
   WHERE carrid IN @lr_carrid
     AND fldate IN @lr_fldate
   ORDER BY carrid, connid, fldate.

  IF gt_data IS INITIAL.
    MESSAGE s001. "Data not found
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left     " Side to Which Control is Docked
      extension = 5000.               " Control Extension

  CREATE OBJECT gcl_splitter
    EXPORTING
      parent  = gcl_container                   " Parent Container
      rows    = 1                   " Number of Rows to be displayed
      columns = 2.                   " Number of Columns to be Displayed

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1                  " Row
      column    = 1                 " Column
    RECEIVING
      container = gcl_container_data.                 " Container

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1                  " Row
      column    = 2                 " Column
    RECEIVING
      container = gcl_container_smoker.                 " Container

  CREATE OBJECT gcl_grid_data
    EXPORTING
      i_parent = gcl_container_data.                  " Parent Container

  CREATE OBJECT gcl_grid_smoker
    EXPORTING
      i_parent = gcl_container_smoker.                  " Parent Container

  CALL METHOD gcl_splitter->set_column_width
    EXPORTING
      id    = 1                   " Row ID
      width = 40.                 " Height

  SET HANDLER : lcl_event_handler=>handle_hotspot_click FOR gcl_grid_data.

  gs_variant-report = sy-repid.
  gs_variant-handle = 'UP'.

  CALL METHOD gcl_grid_data->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant                  " Layout
      i_save          = 'A'                 " Save Layout
      i_default       = 'X'              " Default Display Variant
      is_layout       = gs_layout                " Layout
    CHANGING
      it_outtab       = gt_data                 " Output Table
      it_sort         = gt_sort
      it_fieldcatalog = gt_fcat_data.                 " Field Catalog

  gs_variant-handle = 'DN'.

  CALL METHOD gcl_grid_smoker->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant                  " Layout
      i_save          = 'A'                 " Save Layout
      i_default       = 'X'              " Default Display Variant
      is_layout       = gs_layout                " Layout
    CHANGING
      it_outtab       = gt_smoker                 " Output Table
      it_fieldcatalog = gt_fcat_smoker.                 " Field Catalog

ENDFORM.

FORM get_smoker  USING pv_carrid TYPE scarr-carrid
                       pv_connid TYPE sbook-connid
                       pv_fldate TYPE zc505cds0025-fldate.

  REPLACE ALL OCCURRENCES OF '-' IN pv_fldate WITH ''.

*  pv_fldate = pv_fldate+0(4) && pv_fldate+5(2) && pv_fldate+8(2).

  CALL METHOD zc505cl_amdp_06=>get_data
    EXPORTING
      iv_carrid = pv_carrid
      iv_connid = pv_connid
      iv_fldate = pv_fldate
    IMPORTING
      et_data   = DATA(lt_smoker).

  IF lt_smoker IS NOT INITIAL.

    SORT lt_smoker BY bookid.
    gt_smoker = CORRESPONDING #( lt_smoker ).

  ELSE.

    REFRESH gt_smoker.
    PERFORM refresh_grid.

  ENDIF.

ENDFORM.

FORM make_fcat  USING    pv_key
                         pv_field
                         pv_text
                         pv_ref_table
                         pv_ref_field.

  DATA(gs_fcat) = VALUE lvc_s_fcat(
                          key       = pv_key
                          fieldname = pv_field
                          coltext   = pv_text
                          ref_table = pv_ref_table
                          ref_field = pv_ref_field
                                   ).

  CASE pv_field.
    WHEN 'SMOKER'.
      gs_fcat-checkbox = 'X'.
    WHEN 'LUGGWEIGHT'.
      gs_fcat-qfieldname = 'WUNIT'.
    WHEN 'ICON'.
      gs_fcat-icon = 'X'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat_smoker.

ENDFORM.

FORM handle_hotspot_click  USING   ps_row TYPE lvc_s_row
                                   ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

*  READ TABLE gt_data INTO gs_data INDEX ps_row-index.

  gs_data = VALUE #( gt_data[ ps_row-index ] OPTIONAL ).

  PERFORM get_smoker USING gs_data-carrid
                           gs_data-connid
                           gs_data-fldate.

  IF gt_smoker IS INITIAL.
    MESSAGE s000 WITH gs_data-fldate 'Smoker not found'.
    EXIT.
  ENDIF.

  PERFORM refresh_grid.

ENDFORM.
