* ADBC 통해서 Insert, Update, Delete 등을 수행함

* ADBC 관련
  DATA : lcl_conn    TYPE REF TO cl_sql_connection,
         lcl_sql     TYPE REF TO cl_sql_statement,
         lr_data     TYPE REF TO data,
         lx_sql_exec TYPE REF TO cx_root,
         lv_sql      TYPE string,
         lv_msg      TYPE string.

  DATA : lv_mtart TYPE mara-mtart VALUE 'FERT',
         lv_matkl TYPE mara-matkl VALUE '001',
         lv_matnr TYPE mara-matnr VALUE 'ST_10'.

  TRY.

      lcl_conn = cl_sql_connection=>get_connection( 'HANADB' ).

      CREATE OBJECT lcl_sql
        EXPORTING
          con_ref = lcl_conn.

      lv_sql = | UPDATE ZC526T0004 SET MTART = '{ lv_mtart }', | &&
               |                       MATKL = '{ lv_matkl }'  | &&
               | WHERE MATNR = '{ lv_matnr }'                  |.

* Transaction 일으키는 DML 이므로 execute_update 호출
      lcl_sql->execute_update( lv_sql ).

    CATCH cx_sql_exception INTO lx_sql_exec. "에러 발생시 처리하는곳

      lv_msg = lx_sql_exec->get_text( ). "에러 메시지 받는다

      MESSAGE s000 WITH lv_msg DISPLAY LIKE 'E'.

* Transaction 에 실패 했다면 롤백 시킴
      lcl_conn->rollback( ).
      lcl_conn->close( ).

      EXIT.

  ENDTRY.

  lcl_conn->commit( ).
  lcl_conn->close( ).

ENDFORM.
