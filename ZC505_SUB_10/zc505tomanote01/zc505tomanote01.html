SELECT carrid, carrname, @sy-datum AS today, @lv_name AS name, 'Class 5' AS class,
       CASE carrid
         WHEN 'AA' THEN '아메리카 에어'
         WHEN 'KA' THEN '대한항공'
         WHEN 'LH' THEN url
         WHEN 'UA' THEN
                        CASE currcode
                          WHEN 'USD' THEN '달러'
                          WHEN 'EUR' THEN '유로'
                          ELSE currcode
                        END
         ELSE '기타항공'
       END AS carrname_kr
  INTO TABLE @DATA(lt_scarr)
  FROM scarr.

*FLOOR : 버림, CEIL : 올림
*CAST  : 형변환 : DEC 와 INT2, INT4 만 가능
*CDS View 에서는 CAST 시 데이터엘리먼트 와 빌트인 타입 모두 가능
*CDS View 에서 빌트인 타입 사용 예
*--> CAST( carrname AS abap.char(10) )
*--> CAST( 'AAAA' AS matnr )
*--> CAST( price AS abap.int4 )
*CONCAT_WITH_SPACE( 필드(또는 값), 필드(또는 값), 공백개수 )
SELECT carrid, connid, fldate, price, ( price * 2 ) + 2 AS price_2,
       div( 52, 2 ) AS mok, mod( 9, 2 ) AS remain, abs( price - 1000 ) AS abs,
       floor( price ) AS floor_price, ceil( price ) AS ceil_price,
       CAST( price AS DEC( 10, 2 ) ) AS cast_price_dec,
       CAST( price AS INT4 ) AS cast_price_int,
       concat( carrid, fldate ) AS concat_data,
       concat_with_space( carrid, connid, 2 ) AS with_space
  FROM sflight
  INTO TABLE @DATA(lt_sflight).
