FORM get_data .

  _clear gs_ship gt_ship. "select 전에 clear
  REFRESH : gt_ship_del.   "삭제 후보 테이블 Clear

  DATA : lr_tknum TYPE RANGE OF zc505tship-tknum. "RANGE 변수 사용하여 필수값 입력하지 않아도 값 나오게 설정

  IF pa_tknum IS NOT INITIAL.

    lr_tknum = VALUE #( BASE lr_tknum
                            (
                            low    = pa_tknum
                            sign   = 'I'
                            option = 'EQ'
                              )
                      ).
  ENDIF.

  SELECT tknum, tpnum, vbeln, vbtyp,
         shtyp, signi, tprfo, smeng,
         meins, netwr, waerk
    INTO CORRESPONDING FIELDS OF TABLE @gt_ship
    FROM zc505tship
   WHERE tknum IN @lr_tknum
   ORDER BY tknum, tpnum.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left  " Side to Which Control is Docked
      extension = 5000.                        " Control Extension

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.                " Parent Container

  SET HANDLER : lcl_event_handler=>handle_toolbar          FOR gcl_grid,
                lcl_event_handler=>handle_user_command     FOR gcl_grid,
                lcl_event_handler=>handle_changed_finished FOR gcl_grid.

  CALL METHOD gcl_grid->register_edit_event     "event 가 더 민감하게 받을 수 있게 함수 사용
    EXPORTING
      i_event_id = gcl_grid->mc_evt_modified.   " Event ID

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant       " Layout
      i_save               = 'A'              " Save Layout
      i_default            = 'X'              " Default Display Variant
      is_layout            = gs_layout        " Layout
      it_toolbar_excluding = gt_exclude       " Excluded Toolbar Standard Functions
    CHANGING
      it_outtab            = gt_ship          " Output Table
      it_fieldcatalog      = gt_fcat.         " Field Catalog

ENDFORM.

FORM delete_data .

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = DATA(lt_row).                  " Indexes of Selected Rows

  IF lt_row IS INITIAL.  " 만약 사용자가 행을 선택하지 않고 delete를 눌렀을때 알려주는 안내 메세지
    MESSAGE s000 WITH 'First Select rows'.
    EXIT.
  ENDIF.

  SORT lt_row BY index DESCENDING. " "ITAB 의 ROW 를 지울 때는 역순으로 삭제해야한다. DESENDING 정렬

  LOOP AT lt_row INTO DATA(ls_row).

    gs_ship = VALUE #( gt_ship[ ls_row-index ] OPTIONAL ).

    CHECK gs_ship IS NOT INITIAL.

    APPEND gs_ship TO gt_ship_del.  "삭제할 레코드를 gt__del 에 옮겨준다.
    CLEAR gs_ship.

    DELETE gt_ship INDEX ls_row-index. "gt_mat 에서 해당 레코드를 지워준다.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM save_data .

  DATA : ls_save   TYPE zc505tship,
         lt_save   LIKE TABLE OF ls_save,
         lt_del    TYPE TABLE OF zc505tship,
         lv_commit.

  CALL METHOD gcl_grid->check_changed_data. ""변화 감지

  IF gt_ship_del IS NOT INITIAL.  "삭제 테이블에 있는 데이터를 제거

    lt_del = CORRESPONDING #( gt_ship_del ).

    DELETE zc505tship FROM TABLE lt_del.

    IF sy-dbcnt > 0.  " 0 이상이면 commit 에 'X' 를 넣어준다.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  LOOP AT gt_ship INTO gs_ship WHERE mark EQ 'X'.

    IF gs_ship-tknum IS INITIAL OR gs_ship-tpnum IS INITIAL.

      MESSAGE s000 WITH 'Input key field' DISPLAY LIKE 'E'.
      CONTINUE.                                    "KEY 값이 입력되지 않았을 경우 저장 방지

    ENDIF.

    ls_save = CORRESPONDING zc505tship( gs_ship ). "gs_ship 은 다른 곳에서 사용이 되고 있을수 있다. ls_save 로 옮겨주고 append 실행

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.              "lt_save 에 값이 있으면
    MODIFY zc505tship FROM TABLE lt_save. "DB 테이블에 반영 (Insert or Update)

    IF sy-dbcnt > 0.                      " 0 이상이면 commit 에 'X' 를 넣어준다.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.

    COMMIT WORK AND WAIT.

    REFRESH gt_ship_del.
    PERFORM clear_mark.                   "다음 변경 확인을 위한 표시 해제
    MESSAGE s002.                         "Data was saved successfully

  ELSE.
    ROLLBACK WORK.
    MESSAGE s000 WITH 'Data was not changed' DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM clear_mark .
  "필드 심볼 사용하여 로직 단순화
  LOOP AT gt_ship ASSIGNING FIELD-SYMBOL(<fs_ship>) WHERE mark EQ 'X'.
    "체크(mark='X')된 행만 대상으로 반복
    CLEAR <fs_ship>-mark.

  ENDLOOP.
