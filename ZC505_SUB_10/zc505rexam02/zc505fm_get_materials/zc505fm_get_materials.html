FUNCTION zc505fm_get_materials.
*"----------------------------------------------------------------------
*"*"Local Interface:
*"  IMPORTING
*"     REFERENCE(IV_WERKS) TYPE  T001W-WERKS
*"  EXPORTING
*"     REFERENCE(EV_CODE) TYPE  CHAR1
*"     REFERENCE(EV_MSG) TYPE  CHAR100
*"  TABLES
*"      ET_DATA STRUCTURE  ZC505T0047
*"----------------------------------------------------------------------

*       Global data declarations

  SELECT matnr werks stlan stlnr stlal
         losbs losvn bmein
         erdat erzet ernam
         aedat aezet aenam
    INTO CORRESPONDING FIELDS OF TABLE et_data
    FROM zc505t0047
   WHERE werks EQ iv_werks
   ORDER BY matnr werks stlan stlnr stlal. "KEY 값 기준으로 정렬

  IF sy-subrc NE 0.

  ELSE.

  ENDIF.

ENDFUNCTION.
