FORM save_data .

  DATA : lt_save   TYPE TABLE OF zc505t0047,
         lt_del    TYPE TABLE OF zc505t0047,
         lv_commit.

  CALL METHOD gcl_grid->check_changed_data.

  "삭제 처리 (gt_mat_del -> lt_del)
  IF gt_mat_del IS NOT INITIAL.

    lt_del = CORRESPONDING #( gt_mat_del ).

    DELETE zc505t0047 FROM TABLE lt_del.

    IF sy-subrc EQ 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  LOOP AT gt_mat ASSIGNING FIELD-SYMBOL(<fs_mat>) WHERE mark IS NOT INITIAL.

    IF <fs_mat>-matnr IS INITIAL OR <fs_mat>-werks IS INITIAL OR
       <fs_mat>-stlal IS INITIAL OR <fs_mat>-stlnr IS INITIAL.

      MESSAGE s000 WITH '필수 키 값을 입력하세요.' DISPLAY LIKE 'E'.
      CONTINUE.

    ENDIF.

    DATA(ls_save) = CORRESPONDING zc505t0047( <fs_mat> ). "경로 지정

    IF ls_save-erdat IS INITIAL. " 신규 데이터
      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.
    ELSE.                        " 기존 데이터 수정
      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.
    ENDIF.

    APPEND ls_save TO lt_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.                  "lt_save 에 값이 있으면

    MODIFY zc505t0047 FROM TABLE lt_save.     "DB 테이블에 반영 (Insert or Update)

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.

    COMMIT WORK AND WAIT.            "확정

    REFRESH gt_mat_del.             "삭제 테이블 초기화

    PERFORM clear_mark.             "다음 변경 확인을 위한 표시 해제
    MESSAGE s002.                   "Data was saved successfully

  ELSE.

    ROLLBACK WORK.                   "아닐 시 ROLLBACK , 에러메세지 표시
    MESSAGE s000 WITH 'Data was not changed' DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM handle_user_command  USING    pv_ucomm TYPE sy-ucomm.

  CASE pv_ucomm.

    WHEN 'CREATE'.              "행 추가 버튼
      PERFORM create_data.      "신규 행 추가
      PERFORM refresh_grid.     "ALV 에 반영

    WHEN 'DELETE'.              "삭제 버튼
      PERFORM delete_data.      "선택 행 삭제

  ENDCASE.

ENDFORM.

FORM handle_changed_finished  USING    pv_modified
                                       pt_good_cells TYPE lvc_t_modi.

  DATA ls_modi TYPE lvc_s_modi.

  CHECK pv_modified IS NOT INITIAL. "pv_modified 변경이 감지되면 체크 변경이 이루어지지 않으면 아래 로직을 안탄다.

  LOOP AT pt_good_cells INTO ls_modi.

    READ TABLE gt_mat INTO gs_mat INDEX ls_modi-row_id.

    IF sy-subrc NE 0.
      CONTINUE.
    ENDIF.

    gs_mat-mark = 'X'. "변경된 값에 MARK 를 넣어 SAVE 로직 안 LOOP를 통해 변경될 수 있게 해준다.

    MODIFY gt_mat FROM gs_mat INDEX ls_modi-row_id
    TRANSPORTING mark.   "mark 수정

  ENDLOOP.

  PERFORM refresh_grid. " ALV 화면 갱신

ENDFORM.

*& Form delete_data
*&---------------------------------------------------------------------*

*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM delete_data .

  DATA: lt_row TYPE lvc_t_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = lt_row.

  IF lt_row IS INITIAL.
    MESSAGE s000 WITH '삭제할 행을 먼저 선택하세요.'.
    EXIT.
  ENDIF.

  SORT lt_row BY index DESCENDING.

  LOOP AT lt_row ASSIGNING FIELD-SYMBOL(<fs_row>).

    READ TABLE gt_mat ASSIGNING FIELD-SYMBOL(<fs_mat>) INDEX <fs_row>-index.

    IF sy-subrc EQ 0.
      APPEND <fs_mat> TO gt_mat_del.

      DELETE gt_mat INDEX <fs_row>-index.
    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.
