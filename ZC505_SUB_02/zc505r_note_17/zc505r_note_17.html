REPORT zc505r_note_17 MESSAGE-ID zc505m.

TABLES : ekko.

DATA : BEGIN OF ls_ekko,
         ebeln TYPE ekko-ebeln,
         ebelp TYPE ekpo-ebelp,
         bukrs TYPE ekko-bukrs,
         butxt TYPE t001-butxt, " 텍테
         bstyp TYPE ekko-bstyp,
         bsart TYPE ekko-bsart,
         batxt TYPE t161t-batxt, " 텍테
         matnr TYPE ekpo-matnr,
         maktx TYPE makt-maktx, " 텍테
         werks TYPE ekpo-werks,
         name1 TYPE t001w-name1, " 텍테
         menge TYPE ekpo-menge,
         meins TYPE ekpo-meins,
       END OF ls_ekko,
       lt_ekko  LIKE TABLE OF ls_ekko,

       ls_t001  TYPE t001,
       lt_t001  LIKE TABLE OF ls_t001, "1번
       ls_t161t TYPE t161t,
       lt_t161t LIKE TABLE OF ls_t161t, "2번
       ls_makt  TYPE makt,
       lt_makt  LIKE TABLE OF ls_makt, "3번
       ls_t001w TYPE t001w,
       lt_t001w LIKE TABLE OF ls_t001w, "4번
       lv_tabix TYPE sy-tabix.

CLEAR   : ls_ekko, ls_t001, ls_t161t, ls_makt, ls_t001w, lv_tabix.
REFRESH : lt_ekko, lt_t001, lt_t161t, lt_makt, lt_t001w.

SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME TITLE TEXT-001.

  SELECT-OPTIONS : so_ebeln FOR ekko-ebeln.

SELECTION-SCREEN END OF BLOCK bl1.

SELECT a~ebeln a~bukrs a~bstyp a~bsart
       b~ebelp b~matnr b~werks b~menge b~meins
  INTO CORRESPONDING FIELDS OF TABLE lt_ekko
  FROM ekko AS a
 INNER JOIN ekpo AS b
    ON a~ebeln EQ b~ebeln
 WHERE a~ebeln IN so_ebeln.

IF sy-subrc NE 0.
  MESSAGE s001 DISPLAY LIKE 'E'.
  STOP.
ENDIF.

SELECT bukrs butxt
  INTO CORRESPONDING FIELDS OF TABLE lt_t001
  FROM t001.

SELECT bsart batxt
  INTO CORRESPONDING FIELDS OF TABLE lt_t161t
  FROM t161t
 WHERE spras = sy-langu.

SELECT werks name1
  INTO CORRESPONDING FIELDS OF TABLE lt_T001W
  FROM t001w.

SELECT matnr maktx
  INTO CORRESPONDING FIELDS OF TABLE lt_makt
  FROM makt
 WHERE spras = sy-langu.

LOOP AT lt_ekko INTO ls_ekko.
  lv_tabix = sy-tabix.
*1번
  READ TABLE lt_t001 INTO ls_t001 WITH KEY bukrs = ls_ekko-bukrs.

  IF sy-subrc EQ 0.
    ls_ekko-butxt = ls_t001-butxt.

    MODIFY lt_ekko FROM ls_ekko INDEX lv_tabix
    TRANSPORTING butxt.

  ENDIF.
*2번
  READ TABLE lt_t161t INTO ls_t161t WITH KEY bsart = ls_ekko-bsart.

  IF sy-subrc EQ 0.
    ls_ekko-batxt = ls_t161t-batxt.

    MODIFY lt_ekko FROM ls_ekko INDEX lv_tabix
    TRANSPORTING batxt.

  ENDIF.
*3번
  READ TABLE lt_t001w INTO ls_t001w WITH KEY werks = ls_ekko-werks.

  IF sy-subrc IS INITIAL.
    ls_ekko-name1 = ls_t001w-name1.

    MODIFY lt_ekko FROM ls_ekko INDEX lv_tabix
    TRANSPORTING name1.

  ENDIF.
*4번
  READ TABLE lt_makt INTO ls_makt WITH KEY matnr = ls_ekko-matnr.

  IF sy-subrc EQ 0.
    ls_ekko-maktx = ls_makt-maktx.

    MODIFY lt_ekko FROM ls_ekko INDEX lv_tabix
    TRANSPORTING maktx.

  ENDIF.

ENDLOOP.

cl_demo_output=>display( lt_ekko ).
