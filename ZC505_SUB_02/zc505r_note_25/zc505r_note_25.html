  SELECT-OPTIONS : so_matnr FOR mara-matnr NO INTERVALS,
                   so_mtart FOR mara-mtart NO-EXTENSION,

SELECT a~matnr a~mtart a~pstat a~meins "mara  .
       b~werks b~ekgrp b~dispo   " marc .
       c~lgort c~labst "mard  .
       d~maktx "makt
       e~mtbez "t134t
       f~wgbez60 "t023t
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM mara AS a
 INNER JOIN marc AS b
    ON a~matnr EQ b~matnr
  LEFT OUTER JOIN mard AS c "mard 에 데이터가 하나 밖에 없어서 모든 값을 보이게 하러면 left outer 조인으로 걸어야함 .
    ON b~matnr EQ c~matnr
   AND b~werks EQ c~werks
  LEFT OUTER JOIN makt AS d
    ON a~matnr EQ d~matnr
   AND d~spras EQ sy-langu
  LEFT OUTER JOIN t134t AS e
    ON a~mtart EQ e~mtart
   AND e~spras EQ sy-langu
  LEFT OUTER JOIN t023t AS f
    ON a~matkl EQ f~matkl
   AND f~spras EQ sy-langu
 WHERE a~matnr IN so_matnr
   AND a~matnr IN so_mtart
   AND b~werks IN so_werks.

IF sy-subrc NE 0.
  MESSAGE s001 DISPLAY LIKE 'E'.
  STOP.
ENDIF.

SELECT ekgrp eknam
  INTO CORRESPONDING FIELDS OF TABLE gt_ekgrp
  FROM t024.

SORT gt_ekgrp BY ekgrp.

LOOP AT gt_data.

  gv_tabix = sy-tabix.

  READ TABLE gt_ekgrp WITH KEY ekgrp = gt_data-ekgrp BINARY SEARCH.

  IF sy-subrc EQ 0.
*    gt_data-eknam = gt_ekgrp-eknam.

    MOVE-CORRESPONDING gt_ekgrp to gt_data. "eknam 필드를 옮김 .

    MODIFY gt_data INDEX gv_tabix TRANSPORTING eknam.

  ENDIF.

ENDLOOP.
