REPORT zc505r_note_30 MESSAGE-ID zc505m.

*ZC505V0029

TABLES : sflight.

SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME TITLE TEXT-t01.

  PARAMETERS : pa_carr TYPE sflight-carrid OBLIGATORY DEFAULT 'LH'.
  SELECT-OPTIONS : so_con FOR sflight-connid OBLIGATORY.

SELECTION-SCREEN END OF BLOCK bl1.

DATA : BEGIN OF gs_sflight,
         carrid     TYPE sflight-carrid,
         carrname   TYPE zc505v0029-carrname,
         connid     TYPE sflight-connid,
         fldate     TYPE sflight-fldate,
         cityfrom   TYPE zc505v0029-cityfrom,
         cityto     TYPE zc505v0029-cityto,
         planetype  TYPE sflight-planetype,
         seatsmax_b TYPE sflight-seatsmax_b,
         url        TYPE zc505v0029-url,
       END OF gs_sflight,

       gt_sflight LIKE TABLE OF gs_sflight,

       BEGIN OF gs_sflight2,
         carrid     TYPE sflight-carrid,
         connid     TYPE sflight-connid,
         fldate     TYPE sflight-fldate,
         cityfrom   TYPE zc505v0029-cityfrom,
         cityto     TYPE zc505v0029-cityto,
         planetype  TYPE sflight-planetype,
         seatsmax_b TYPE sflight-seatsmax_b,
       END OF gs_sflight2,

       gt_sflight2 LIKE TABLE OF gs_sflight2.

CLEAR   : gs_sflight, gs_sflight2.
REFRESH : gt_sflight, gt_sflight2.

SELECT a~carrid a~connid a~fldate a~planetype a~seatsmax_b
       b~carrname b~cityfrom b~cityto b~url
  INTO CORRESPONDING FIELDS OF TABLE gt_sflight
  FROM sflight AS a
  INNER JOIN zc505v0029 AS b
  ON a~carrid    EQ b~carrid
  AND a~connid   EQ b~connid
  WHERE a~carrid EQ pa_carr
  AND  a~connid  IN so_con.
*from 절에 뷰가 오는게 맞음.

IF sy-subrc NE 0.
  MESSAGE s001.
  STOP.
ENDIF.

LOOP AT gt_sflight INTO gs_sflight.

  CASE gs_sflight-seatsmax_b.

    WHEN '30'.
      gs_sflight-planetype = '747-400'.
    WHEN '21'.
      gs_sflight-planetype = 'A340-600'.
    WHEN OTHERS.
      CLEAR : gs_sflight-planetype.
  ENDCASE.

  MOVE-CORRESPONDING gs_sflight TO gs_sflight2.
  APPEND gs_sflight2 TO  gt_sflight2.

  CLEAR gs_sflight2.

ENDLOOP.

cl_demo_output=>display( gt_sflight2 ).
