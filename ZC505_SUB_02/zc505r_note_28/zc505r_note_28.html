SELECT a~matnr a~mtart a~matkl
       b~werks b~dispo b~ekgrp
  INTO CORRESPONDING FIELDS OF TABLE gt_data
  FROM mara AS a
 INNER JOIN marc AS b
    ON  a~matnr EQ b~matnr
 WHERE a~matnr IN so_mat
   AND b~werks EQ pa_wer.

IF sy-subrc NE 0.
  MESSAGE s001.
  STOP.
ENDIF.

SELECT matnr maktx
  INTO CORRESPONDING FIELDS OF TABLE gt_makt
  FROM makt
 WHERE spras = sy-langu.

SELECT matkl wgbez
  INTO CORRESPONDING FIELDS OF TABLE gt_t023t
  FROM t023t
 WHERE spras = sy-langu.

SELECT dispo dsnam
  INTO CORRESPONDING FIELDS OF TABLE gt_t024d
  FROM t024d
  WHERE werks EQ pa_wer.

SELECT ekgrp eknam
  INTO CORRESPONDING FIELDS OF TABLE gt_t024
  FROM t024.

SORT : gt_makt  BY matnr,
       gt_t023t BY matkl,
       gt_t024d BY dispo,
       gt_t024  BY ekgrp.

  IF sy-subrc EQ 0.

  ENDIF.

  MOVE-CORRESPONDING gt_data TO gt_data2.
