REPORT zc505r_note_18 MESSAGE-ID zc505m.


DATA : BEGIN OF ls_bkpf,
         bukrs          TYPE bkpf-bukrs,
         belnr          TYPE bkpf-belnr,
         buzei          TYPE bseg-buzei,
         gjahr          TYPE bkpf-gjahr,
         blart          TYPE bkpf-blart,
         ltext          TYPE t003t-ltext, "텍스트 테이블
         budat          TYPE bkpf-budat,
         bldat          TYPE bkpf-bldat,
         bschl          TYPE bseg-bschl,
         koart          TYPE bseg-koart,
         koart_text(50),
         shkzg          TYPE bseg-shkzg,
         shkzg_text(20),
         dmbtr          TYPE bseg-dmbtr,
       END OF ls_bkpf,

       lt_bkpf  LIKE TABLE OF ls_bkpf,
       lv_tabix TYPE sy-tabix.

*테이블, 변수 클리어
CLEAR   : ls_bkpf, lv_tabix.
REFRESH : lt_bkpf.

SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME TITLE TEXT-t01.

  PARAMETERS : pa_gjahr TYPE bkpf-gjahr OBLIGATORY DEFAULT '2010'."필수 입력, 기본값 '2010' 으로 설정

SELECTION-SCREEN END OF BLOCK bl1.

*INITIALIZATION. "초기값 설정
*  pa_gjahr = '2010'.
*
*START-OF-SELECTION.

  SELECT a~bukrs a~belnr a~gjahr a~blart a~budat a~bldat "bkpf 테이블
         b~buzei b~bschl b~koart b~shkzg b~dmbtr
    INTO CORRESPONDING FIELDS OF TABLE lt_bkpf
    FROM bkpf AS a
   INNER JOIN bseg AS b "HEADER 을 LEFT로 조인 실행
      ON a~bukrs EQ b~bukrs
     AND a~belnr EQ b~belnr
     AND a~gjahr EQ b~gjahr
    LEFT OUTER JOIN t003t AS c
      ON a~blart EQ c~blart
   WHERE a~gjahr EQ pa_gjahr.

  IF sy-subrc NE 0.
    MESSAGE s001 DISPLAY LIKE 'E'.
    STOP.
  ENDIF.

  LOOP AT lt_bkpf INTO ls_bkpf.

    lv_tabix = sy-tabix.

    CASE ls_bkpf-koart.
      WHEN 'A'.
        ls_bkpf-koart_text = 'Asset'.
      WHEN 'D'.
        ls_bkpf-koart_text = 'Customers'.
      WHEN 'K'.
        ls_bkpf-koart_text = 'Suppliers'.
      WHEN 'M'.
        ls_bkpf-koart_text = 'Material'.
      WHEN 'S'.
        ls_bkpf-koart_text = 'G/L Account'.
    ENDCASE.
    "shkzg 가 H, S 일때 아래의 값으로 변경
    CASE ls_bkpf-shkzg.
      WHEN 'H'.
        ls_bkpf-shkzg_text = 'Credit'.
      WHEN 'S'.
        ls_bkpf-shkzg_text = 'Debit'.
    ENDCASE.

    MODIFY lt_bkpf FROM ls_bkpf INDEX lv_tabix
    TRANSPORTING koart_text shkzg_text.

  ENDLOOP.

  cl_demo_output=>display( lt_bkpf ).
