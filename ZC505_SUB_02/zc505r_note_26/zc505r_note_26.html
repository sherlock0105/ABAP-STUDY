REPORT zc505r_note_26 MESSAGE-ID zc505m.

TABLES : sbuspart.

SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME TITLE TEXT-t01.

  PARAMETERS: pa_bus TYPE sbuspart-buspartnum OBLIGATORY DEFAULT '00000061'.
  SELECT-OPTIONS : so_bus FOR sbuspart-contact NO INTERVALS.

  PARAMETERS : pa_ta RADIOBUTTON GROUP rb1 DEFAULT 'X',
               pa_fc RADIOBUTTON GROUP rb1.

SELECTION-SCREEN END OF BLOCK bl1.

DATA : ls_sbus  TYPE sbuspart,
       lt_sbus  LIKE TABLE OF ls_sbus,
       lv_radio TYPE sbuspart-buspatyp.

CASE 'X'.

  WHEN pa_ta.
    lv_radio = 'TA'.
  WHEN pa_fc.
    lv_radio = 'FC'.

ENDCASE.

SELECT buspartnum contact contphono buspatyp
  INTO CORRESPONDING FIELDS OF TABLE lt_sbus
  FROM sbuspart
 WHERE buspartnum EQ pa_bus
   AND contact    IN so_bus
   AND buspatyp   EQ lv_radio.

IF sy-subrc NE space.
  MESSAGE s001.
  STOP.
ENDIF.

cl_demo_output=>display( lt_sbus ).
