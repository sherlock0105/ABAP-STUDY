CLEAR   : ls_sbook.  "구조체 내부값 모두 초기화
REFRESH : lt_sbook. "내부테이블 모든 행 삭제

* [3] SELECT 실행 → lt_flight 에 여러 레코드 로드
SELECT carrid connid fldate bookid customid
       custtype smoker invoice
  INTO CORRESPONDING FIELDS OF TABLE lt_sbook
  FROM sbook
 WHERE carrid   EQ 'LH'
   AND connid   IN ( '0401', '2402' )
   AND customid GE '4387'.

IF sy-subrc NE 0.

  MESSAGE s001 DISPLAY LIKE 'E'.
  STOP.

ENDIF.

LOOP AT lt_sbook INTO ls_sbook.

  lv_tabix = sy-tabix.

  CASE abap_true.

    WHEN ls_sbook-invoice OR ls_sbook-smoker.

      ls_sbook-billing = 'A'.

      MODIFY lt_sbook FROM ls_sbook INDEX lv_tabix
      TRANSPORTING billing.

  ENDCASE.

ENDLOOP.

cl_demo_output=>display( lt_sbook ).

SELECT carrid connid fldate bookid customid
       custtype smoker invoice class
  INTO CORRESPONDING FIELDS OF TABLE lt_sbook2
  FROM sbook
 WHERE carrid     = 'DL'
   AND custtype   = 'P'
   AND order_date = '20251024'.

IF sy-subrc NE 0.

  MESSAGE s001 DISPLAY LIKE 'E'.
  STOP.

ENDIF.

LOOP AT lt_sbook2 INTO ls_sbook2.

  lv_tabix2 = sy-tabix.

  IF ls_sbook2-smoker = abap_true AND ls_sbook2-invoice = abap_true.

    ls_sbook2-class = 'F'.

    MODIFY lt_sbook2 FROM ls_sbook2 INDEX lv_tabix2
    TRANSPORTING class.
