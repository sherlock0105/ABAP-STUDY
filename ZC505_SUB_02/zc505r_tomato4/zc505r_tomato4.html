DATA :BEGIN OF gs_group,
        group_id        TYPE zc505t0029-group_id,
        group_txt       TYPE zc505t0029-group_txt,
        member_id       TYPE zc505t0040-member_id,
        member_name     TYPE zc505t0040-member_name,
        member_gender   TYPE zc505t0040-member_gender,
        member_addr     TYPE zc505t0040-member_addr,
        member_phone    TYPE zc505t0040-member_phone,
        group_url       TYPE zc505t0029-group_url,
        gender_text(10),
      END OF gs_group,

      gt_group LIKE TABLE OF gs_group,
      lv_tabix TYPE sy-tabix.

SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME TITLE TEXT-001.

  SELECT-OPTIONS : so_grp FOR zc505t0029-group_id NO-EXTENSION NO INTERVALS.

  PARAMETERS : pa_mem TYPE zc505t0040-member_id.

SELECTION-SCREEN END OF BLOCK bl1.


SELECT a~group_id a~group_txt a~group_url
       b~member_id b~member_name b~member_gender b~member_addr b~member_phone
  INTO CORRESPONDING FIELDS OF TABLE gt_group
  FROM zc505t0029 AS a
 INNER JOIN zc505t0040 AS b
    ON a~group_id  EQ b~group_id
 WHERE a~group_id  IN so_grp "빈값으로 넣게되면 WHERE 조건에서 사라짐
   AND b~member_id = pa_mem
  ORDER BY b~member_id.

"index 를 활용해서 테이블을 읽는다.
LOOP AT gt_group INTO gs_group.
"gs_group 데이터를 읽고 변환시켜줌.
  lv_tabix = sy-tabix.

  IF gs_group-member_gender = 'M'.
    gs_group-gender_text   = '남자'.
"modify 업데이트, loop 안에서만 바뀐 값을 밖으로 내보내준다.
    MODIFY gt_group FROM gs_group INDEX lv_tabix
    TRANSPORTING gender_text.

  ELSEIF gs_group-member_gender = 'F'.
    gs_group-gender_text   = '여자'.

    MODIFY gt_group FROM gs_group INDEX lv_tabix
    TRANSPORTING gender_text.

  ENDIF.

ENDLOOP.

IF  sy-subrc NE 0.
  "데이터가 없는데도 조회되는게 싫음. 아래 로직을 타지 않게 STOP 을 적어준다.
  "내가 조회하는 데이터가 없을때!.
  MESSAGE s000 WITH '값을 입력하세요' DISPLAY LIKE 'E'.
  STOP.
ENDIF.
