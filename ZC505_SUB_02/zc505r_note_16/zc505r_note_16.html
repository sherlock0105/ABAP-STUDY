REPORT zc505r_note_16 MESSAGE-ID zc505m.

TABLES : ekko.

SELECTION-SCREEN BEGIN OF BLOCK bl1 WITH FRAME TITLE TEXT-t01.

  SELECT-OPTIONS : so_ebeln FOR ekko-ebeln.

SELECTION-SCREEN END OF BLOCK bl1.

DATA : BEGIN OF ls_ekko,
         ebeln TYPE ekko-ebeln,
         ebelp TYPE ekpo-ebelp,
         bukrs TYPE ekko-bukrs,
         butxt TYPE t001-butxt, " 텍테
         bstyp TYPE ekko-bstyp,
         bsart TYPE ekko-bsart,
         batxt TYPE t161t-batxt, " 텍테
         matnr TYPE ekpo-matnr,
         maktx TYPE makt-maktx, " 텍테
         werks TYPE ekpo-werks,
         name1 TYPE t001w-name1, " 텍테
         menge TYPE ekpo-menge,
         meins TYPE ekpo-meins,
       END OF ls_ekko,
       lt_ekko LIKE TABLE OF ls_ekko.

CLEAR   : ls_ekko.
REFRESH : lt_ekko.

SELECT a~ebeln a~bukrs a~bstyp a~bsart
       b~ebelp b~matnr b~werks b~menge b~meins
       c~butxt
       d~batxt
       e~maktx
       f~name1
  INTO CORRESPONDING FIELDS OF TABLE lt_ekko
  FROM ekko AS a
 INNER JOIN ekpo AS b
    ON a~ebeln EQ b~ebeln
 INNER JOIN t001 AS c
    ON a~bukrs EQ c~bukrs
  LEFT OUTER JOIN t161t AS d
    ON a~bsart EQ d~bsart
   AND a~bstyp EQ d~bstyp
   AND d~spras EQ sy-langu
  LEFT OUTER JOIN makt AS e
    ON b~matnr EQ e~matnr
   AND e~spras EQ sy-langu
 INNER JOIN t001w AS f
    ON b~werks EQ f~werks
 WHERE a~ebeln IN so_ebeln.


IF sy-subrc NE 0.
  MESSAGE s001 DISPLAY LIKE 'E'.
  STOP.
ENDIF.

cl_demo_output=>display( lt_ekko ).
