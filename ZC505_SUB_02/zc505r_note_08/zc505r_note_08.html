DATA : ls_scarr TYPE scarr,
       lt_scarr LIKE TABLE OF ls_scarr.

REFRESH : lt_scarr.

SELECT mandt carrid carrname url currcode
  INTO CORRESPONDING FIELDS OF TABLE lt_scarr
  FROM scarr
 WHERE carrid   = 'LH'
   AND currcode = 'EUR'.

IF sy-subrc NE 0.

  WRITE : 'Data not found'.
  STOP.

ENDIF.

*DATA : ls_flight TYPE sflight,
*       lt_flight LIKE TABLE OF ls_flight,
*       lv_tabix  TYPE sy-tabix.

DATA : BEGIN OF ls_flight,
         carrid    TYPE sflight-carrid,
         connid    TYPE sflight-connid,
         fldate    TYPE sflight-fldate,
         price     TYPE sflight-price,
         currency  TYPE sflight-currency,
         planetype TYPE sflight-planetype,
         seatsocc  TYPE sflight-seatsocc,
       END OF ls_flight,

       lt_flight LIKE TABLE OF ls_flight,
       lv_tabix  TYPE sy-tabix.

CLEAR   : ls_flight.
REFRESH : lt_flight.

SELECT carrid connid fldate price currency
  INTO CORRESPONDING FIELDS OF TABLE lt_flight
  FROM sflight.

IF sy-subrc NE 0.

  WRITE : 'Date not found'.
  STOP.

ENDIF.

LOOP AT lt_flight INTO ls_flight.

  lv_tabix = sy-tabix.

  IF ls_flight-carrid = 'AA'.

    ls_flight-carrid = 'KO'.

    MODIFY lt_flight FROM ls_flight INDEX lv_tabix
    TRANSPORTING carrid.

  ENDIF.

  CASE ls_flight-carrid.

    WHEN 'KO'.
      ls_flight-carrid = 'AA'.

      MODIFY lt_flight FROM ls_flight INDEX lv_tabix
      TRANSPORTING carrid.

  ENDCASE.

ENDLOOP.

SELECT carrid connid fldate price currency
  INTO CORRESPONDING FIELDS OF TABLE lt_flight
  FROM sflight.
