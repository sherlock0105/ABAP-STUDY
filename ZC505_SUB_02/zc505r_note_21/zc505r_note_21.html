REPORT zc505r_note_21 MESSAGE-ID zc505m.

DATA : ls_sflight TYPE sflight,
       lt_sflight LIKE TABLE OF ls_sflight,

       BEGIN OF ls_sflight2,
         carrid   TYPE scarr-carrid,
         connid   TYPE sflight-connid,
         fldate   TYPE sflight-fldate,
         carrname TYPE scarr-carrname,
         price    TYPE sflight-price,
         currency TYPE sflight-currency,
         url      TYPE scarr-url,
       END OF ls_sflight2,

       lt_sflight2 LIKE TABLE OF ls_sflight2,
       lv_tabix    TYPE sy-tabix.

CLEAR   : ls_sflight, ls_sflight2, lv_tabix.
REFRESH : lt_sflight, lt_sflight2.

SELECT carrid connid fldate price currency
       planetype seatsmax seatsocc paymentsum
       seatsmax_b seatsocc_b seatsmax_f seatsocc_f
  INTO CORRESPONDING FIELDS OF TABLE lt_sflight
  FROM sflight
 WHERE carrid IN ('LH', 'DL').

IF sy-subrc NE 0.
  MESSAGE s001 DISPLAY LIKE 'E'.
  STOP.
ENDIF.

MOVE-CORRESPONDING lt_sflight TO lt_sflight2.


LOOP AT lt_sflight2 INTO ls_sflight2.

  lv_tabix = sy-tabix.

  CASE ls_sflight2-carrid.

    WHEN 'LH'.
      ls_sflight2-currency = '유로'.
      MODIFY lt_sflight2 FROM ls_sflight2 INDEX lv_tabix
      TRANSPORTING currency.

  ENDCASE.

ENDLOOP.

*cl_demo_output=>display( lt_sflight ).
*cl_demo_output=>display( lt_sflight2 ).

         url      TYPE scarr-url,

         carrname TYPE scarr-carrname,
         url      TYPE scarr-url,
       END OF gs_sflight2,
       gt_sflight2 LIKE TABLE OF gs_sflight2.

SELECT carrid connid airpfrom
       airpto deptime arrtime
INTO CORRESPONDING FIELDS OF TABLE gt_sflight
  FROM spfli.

SELECT carrid carrname url
 INTO CORRESPONDING FIELDS OF TABLE gt_sflight2
 FROM scarr.


LOOP AT gt_sflight INTO gs_sflight.

  lv_tabix = sy-tabix.

  READ TABLE gt_sflight2 INTO gs_sflight2 WITH KEY carrid = gs_sflight-carrid.

  IF sy-subrc EQ 0.

    MOVE-CORRESPONDING gs_sflight2 to gs_sflight.
*    gs_sflight-carrname = gs_sflight2-carrname.
*    gs_sflight-url      = gs_sflight2-url.
    MODIFY gt_sflight FROM gs_sflight INDEX lv_tabix
    TRANSPORTING carrname url.

  ENDIF.

ENDLOOP.
