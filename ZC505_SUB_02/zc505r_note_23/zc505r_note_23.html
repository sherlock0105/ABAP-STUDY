  INTO CORRESPONDING FIELDS OF TABLE gt_scarr_h
  FROM scarr.

  MESSAGE s001.
  STOP.

*'KA' 로 바꾸고 carrname 을 'Korean Air'로 바꿀것

LOOP AT gt_scarr_h.

  lv_tabix = sy-tabix.

  CASE gt_scarr_h-carrid.
    WHEN 'JL'.
      gt_scarr_h-carrid = 'KA'.
      gt_scarr_h-carrname = 'Korean Air'.

      MODIFY gt_scarr_h INDEX lv_tabix TRANSPORTING carrid carrname.
  ENDCASE.

ENDLOOP.
