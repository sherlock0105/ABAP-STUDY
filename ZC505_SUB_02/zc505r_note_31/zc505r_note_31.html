REPORT zc505r_note_31.
*ZC505V0030

DATA : gs_sflight TYPE zc505v0030,
       gt_sflight TYPE TABLE OF zc505v0030.

DATA : BEGIN OF gs_sflight2,
         carrid    TYPE zc505v0030-carrid,
         connid    TYPE zc505v0030-connid,
         fldate    TYPE zc505v0030-fldate,
         planetype TYPE zc505v0030-planetype,
         price     TYPE zc505v0030-price,
         currency  TYPE zc505v0030-currency,
         url       TYPE zc505v0030-url,
       END OF gs_sflight2,
       gt_sflight2 LIKE TABLE OF gs_sflight2.

SELECT carrid connid fldate carrname price currency
       planetype paymentsum seatsmax_b url
INTO CORRESPONDING FIELDS OF TABLE gt_sflight
FROM zc505v0030
WHERE carrid = 'UA'
AND planetype IN ('767-200' ,'A380-800').


LOOP AT gt_sflight INTO gs_sflight.

  CASE gs_sflight-fldate.
    WHEN '20251128'.
      gs_sflight-planetype = '757F'.
    WHEN '20251228'.
      gs_sflight-planetype = 'L-1011-100'.
  ENDCASE.

  MOVE-CORRESPONDING gs_sflight TO gs_sflight2.
  APPEND gs_sflight2 TO gt_sflight2.
  CLEAR : gs_sflight.

ENDLOOP.
