SELECT a~carrid a~connid a~fldate a~planetype a~currency
       b~bookid b~customid b~custtype b~class b~agencynum
  INTO CORRESPONDING FIELDS OF TABLE gt_sflight
  FROM sflight AS a
  INNER JOIN sbook AS b
  ON a~carrid EQ b~carrid
  AND a~connid EQ b~connid
  AND a~fldate EQ b~fldate
  WHERE a~carrid EQ pa_carr
  AND   a~connid IN so_conin
  AND   b~bookid IN so_book
  AND   a~planetype EQ pa_plan
  AND   b~custtype EQ 'P'.
