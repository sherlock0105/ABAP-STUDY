FORM get_data .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~bukrs a~belnr a~blart a~bktxt
         b~buzei b~bschl b~shkzg b~dmbtr b~pswsl b~kokrs
         c~ltext
         d~bezei AS kotext
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM bkpf AS a
   INNER JOIN bseg AS b
      ON a~bukrs EQ b~bukrs
     AND a~belnr EQ b~belnr
     AND a~gjahr EQ b~gjahr
    LEFT OUTER JOIN t003t AS c
      ON a~blart EQ c~blart
     AND c~spras EQ sy-langu
    LEFT OUTER JOIN tka01 AS d
      ON b~kokrs EQ d~kokrs
   WHERE a~bukrs EQ pa_bukrs
     AND a~gjahr EQ pa_gjahr
     AND a~belnr IN so_belnr.

  IF sy-subrc NE 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

*& -->  p1        text

*&---------------------------------------------------------------------*
FORM make_fcat USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.

    WHEN 'DMBTR'.
      gs_fcat-cfieldname = 'PSWSL'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_list .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
      is_variant      = gs_variant
    CHANGING
      it_outtab       = gt_data
      it_fieldcatalog = gt_fcat.

ENDFORM.
