FORM get_sflight .

  REFRESH : gt_sflight.

  SELECT carrid connid fldate price
         currency planetype paymentsum
    INTO CORRESPONDING FIELDS OF TABLE gt_sflight
    FROM sflight
   WHERE carrid IN so_carr.

  IF sy-subrc NE 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.
ENDFORM.

FORM make_fcat .

  gs_fcat-fieldname = 'CARRID'.
  gs_fcat-ref_table = 'SFLIGHT'.
  gs_fcat-key       = 'X'.
  gs_fcat-coltext   = 'Airline'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

  gs_fcat-fieldname = 'CONNID'.
  gs_fcat-ref_table = 'SFLIGHT'.
  gs_fcat-key       = 'X'.
  gs_fcat-coltext   = 'Flight Number'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

  gs_fcat-fieldname = 'FLDATE'.
  gs_fcat-ref_table = 'SFLIGHT'.
  gs_fcat-key       = 'X'.
  gs_fcat-coltext   = 'Flight Date'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

  gs_fcat-fieldname = 'PRICE'.
  gs_fcat-ref_table = 'SFLIGHT'.
  gs_fcat-coltext   = 'Airfare'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

  gs_fcat-fieldname = 'CURRENCY'.
  gs_fcat-ref_table = 'SFLIGHT'.
  gs_fcat-coltext   = 'Currency'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

  gs_fcat-fieldname = 'PLANETYPE'.
  gs_fcat-ref_table = 'SFLIGHT'.
  gs_fcat-coltext   = 'Plane Type'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

  gs_fcat-fieldname = 'PAYMENTSUM'.
  gs_fcat-ref_table = 'SFLIGHT'.
  gs_fcat-coltext   = 'Total'.

  APPEND gs_fcat TO gt_fcat.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_sflight
      it_fieldcatalog = gt_fcat.

ENDFORM.
