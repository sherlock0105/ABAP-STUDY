FORM get_data .

  CLEAR   : gs_scarr.
  REFRESH : gt_scarr.

  SELECT carrid carrname currcode url
    INTO CORRESPONDING FIELDS OF TABLE gt_scarr
    FROM scarr
   WHERE carrid IN so_carr
   ORDER BY carrid.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING. "selection Screen 안에서만 쓸 수 있음.
  ENDIF.

ENDFORM.

FORM data_loop .

  DATA : lv_tabix TYPE sy-tabix.

  LOOP AT gt_scarr INTO gs_scarr.

    lv_tabix = sy-tabix.

    CASE gs_scarr-carrid.
      WHEN 'JL'.
        gs_scarr-carrid   = 'KA'.
        gs_scarr-carrname = 'Korean Air'.
        gs_scarr-currcode = 'KRW'.
        gs_scarr-url      = 'www.korean-air.co.kr'.

        MODIFY gt_scarr FROM gs_scarr INDEX lv_tabix
        TRANSPORTING carrid carrname currcode url.
    ENDCASE.

  ENDLOOP.

ENDFORM.
