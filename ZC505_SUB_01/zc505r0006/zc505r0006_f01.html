FORM get_group .

  REFRESH : gt_group.

  SELECT a~group_id a~group_txt a~group_url
         b~member_id b~member_name b~member_gender
    INTO CORRESPONDING FIELDS OF TABLE gt_group
    FROM zc505t0029 AS a
   INNER JOIN zc505t0040 AS b
      ON a~group_id EQ b~group_id
   WHERE a~group_id IN so_group
   ORDER BY b~member_id.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_structure_name = 'ZC505S0009'
      i_save           = 'A'
      i_default        = 'X'
      is_layout        = gs_layout
    CHANGING
      it_outtab        = gt_group.

ENDFORM.
