FORM get_data .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~bukrs a~gjahr a~belnr a~blart a~budat
         b~buzei b~bschl b~koart b~shkzg b~dmbtr b~pswsl
         c~ltext AS blart_txt
         d~ltext AS bschl_txt
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM bkpf AS a
   INNER JOIN bseg AS b
      ON a~bukrs EQ b~bukrs
     AND a~belnr EQ b~belnr
     AND a~gjahr EQ b~gjahr
    LEFT OUTER JOIN t003t AS c
      ON  a~blart EQ c~blart
     AND c~spras EQ sy-langu
    LEFT OUTER JOIN tbslt AS d
      ON b~bschl EQ d~bschl
     AND b~umskz EQ d~umskz
     AND d~spras EQ sy-langu
   WHERE a~bukrs IN so_bukrs
     AND a~gjahr IN so_gjahr.

  IF sy-subrc NE 0.
    MESSAGE s001 DISPLAY LIKE 'E'.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM data_loop .

  DATA: lv_tabix TYPE sy-tabix.

  LOOP AT gt_data INTO gs_data.

    CLEAR : gs_data.

    lv_tabix = sy-tabix.

    CASE gs_data-shkzg.
      WHEN 'H'.
        gs_data-dmbtr *= -1.

        MODIFY gt_data FROM gs_data INDEX lv_tabix TRANSPORTING dmbtr.

    ENDCASE.

  ENDLOOP.

ENDFORM.
