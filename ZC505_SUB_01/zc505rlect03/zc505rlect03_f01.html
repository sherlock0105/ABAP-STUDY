FORM get_data .

  CLEAR   : gt_mat.
  REFRESH : gt_mat.

  SELECT a~matnr a~mtart a~matkl
         b~werks b~dispo b~ekgrp
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM mara AS a
   INNER JOIN marc AS b
      ON a~matnr EQ b~matnr
   WHERE a~matnr IN so_matnr
     AND b~werks EQ pa_werks.

  IF sy-subrc NE 0.
    MESSAGE s001.
    STOP.
  ENDIF.

ENDFORM.

FORM data_desc .

  CLEAR   : gt_makt, gt_t023t, gt_t024d, gt_t024.
  REFRESH : gt_makt, gt_t023t, gt_t024d, gt_t024.

  SELECT matnr maktx
    INTO CORRESPONDING FIELDS OF TABLE gt_makt
    FROM makt
   WHERE spras = sy-langu.

  SELECT matkl wgbez
    INTO CORRESPONDING FIELDS OF TABLE gt_t023t
    FROM t023t
   WHERE spras = sy-langu.

  SELECT dispo dsnam
    INTO CORRESPONDING FIELDS OF TABLE gt_t024d
    FROM t024d
   WHERE werks EQ pa_werks.

  SELECT ekgrp eknam
    INTO CORRESPONDING FIELDS OF TABLE gt_t024
    FROM t024.

ENDFORM.

FORM get_data2 .

  SORT : gt_makt  BY matnr,
         gt_t023t BY matkl,
         gt_t024D BY dispo,
         gt_t024  BY ekgrp.

  LOOP AT gt_mat.

    MOVE-CORRESPONDING gt_mat TO gt_mat2.

    READ TABLE gt_makt WITH KEY matnr = gt_mat2-matnr BINARY SEARCH.

    IF sy-subrc EQ 0.
      gt_mat2-maktx = gt_makt-maktx.
      CLEAR: gt_makt.
    ENDIF.

    READ TABLE gt_t023t WITH KEY matkl = gt_mat2-matkl BINARY SEARCH.

    IF sy-subrc EQ 0.
      gt_mat2-wgbez = gt_t023t-wgbez.
      CLEAR: gt_t023t.
    ENDIF.

    READ TABLE gt_t024d WITH KEY dispo = gt_mat2-dispo BINARY SEARCH.

    IF sy-subrc EQ 0.
      gt_mat2-dsnam = gt_t024d-dsnam.
      CLEAR: gt_t024d.
    ENDIF.

    READ TABLE gt_t024 WITH KEY ekgrp = gt_mat2-ekgrp BINARY SEARCH.

    IF sy-subrc EQ 0.
      gt_mat2-eknam = gt_t024-eknam.
      CLEAR: gt_t024.
    ENDIF.

    APPEND : gt_mat2.
    CLEAR  : gt_mat2.

  ENDLOOP.

ENDFORM.
