FORM get_data .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT  a~blart a~budat  "bkpf header
          b~belnr b~buzei b~bschl b~koart "bseg Segment
          b~gsber b~shkzg b~dmbtr b~pswsl
          c~ltext
          d~gtext
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM bkpf AS a
   INNER JOIN bseg AS b
      ON a~bukrs EQ b~bukrs
     AND a~belnr EQ b~belnr
     AND a~gjahr EQ b~gjahr
    LEFT OUTER JOIN t003t AS c
      ON a~blart EQ c~blart
     AND c~spras EQ sy-langu
    LEFT OUTER JOIN tgsbt AS d
      ON b~gsber EQ d~gsber
     AND d~spras EQ sy-langu
   WHERE a~bukrs EQ pa_burks
     AND a~gjahr EQ pa_gjahr
     AND a~belnr IN so_belnr.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM makefcat  USING pv_key
                     pv_field
                     pv_text
                     pv_ref_table
                     pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'STATUS'.
      gs_fcat-icon = 'X'.
    WHEN 'DMBTR'.
      gs_fcat-cfieldname = 'PSWSL'.
      gs_fcat-do_sum     = 'X'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.


ENDFORM.

*& <--  p2        text
*&---------------------------------------------------------------------*
FORM set_icon .

  DATA : lv_tabix TYPE sy-tabix,
         ls_color TYPE lvc_s_scol.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    CASE gs_data-shkzg.
      WHEN 'S'.
        gs_data-status = icon_positive.

      WHEN 'H'.
        gs_data-status     = icon_negative.
        ls_color-fname     = 'DMBTR'.
        ls_color-color-col = 6.
        ls_color-color-int = 1.
        ls_color-color-inv = 0.

        APPEND ls_color TO gs_data-color.
        CLEAR  ls_color.

    ENDCASE.

    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING status color.

  ENDLOOP.
ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_data
      it_fieldcatalog = gt_fcat
      it_sort         = gt_sort.

ENDFORM.
