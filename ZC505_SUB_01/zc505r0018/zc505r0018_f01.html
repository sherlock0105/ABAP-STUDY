FORM get_data .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~ebeln a~bukrs a~bstyp a~bsart
         b~ebelp b~matnr b~werks b~menge b~meins
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM ekko AS a
   INNER JOIN ekpo AS b
      ON a~ebeln EQ b~ebeln
   WHERE a~bukrs EQ pa_bukrs
     AND a~ebeln IN so_ebeln.

  IF sy-subrc NE 0.

    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM make_fcat USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

FORM get_text .

  CLEAR   : gs_t161t, gs_makt.
  REFRESH : gt_t161t, gt_makt.

  SELECT bsart bstyp batxt
    INTO CORRESPONDING FIELDS OF TABLE gt_t161t
    FROM t161t
   WHERE spras EQ sy-langu.

  SELECT matnr maktx
    INTO CORRESPONDING FIELDS OF TABLE gt_makt
    FROM makt
   WHERE spras EQ sy-langu.

ENDFORM.

FORM move_data .

  DATA : lv_tabix TYPE sy-tabix.

  LOOP AT gt_data INTO gs_data.

    MOVE-CORRESPONDING gs_data TO gs_data2.

    lv_tabix = sy-tabix.

    READ TABLE gt_t161t INTO gs_t161t WITH KEY bsart = gs_data2-bsart
                                               bstyp = gs_data2-bstyp.

    IF sy-subrc EQ 0.
      gs_data2-batxt = gs_t161t-batxt.
    ENDIF.

    READ TABLE gt_makt INTO gs_makt WITH KEY matnr = gs_data2-matnr.

    IF sy-subrc EQ 0.
      gs_data2-maktx = gs_makt-maktx.
    ENDIF.

    APPEND gs_data2 TO gt_data2.
    CLEAR  gs_data2.

  ENDLOOP.

ENDFORM.

FORM display_list .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_data2
      it_fieldcatalog = gt_fcat.

ENDFORM.
