FORM get_data .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~bukrs a~gjahr a~belnr a~blart a~budat
         b~buzei b~bschl b~koart b~shkzg b~fcsl
         c~ltext AS bltxt
         d~ltext AS bstxt
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM bkpf AS a
   INNER JOIN bseg AS b
      ON a~bukrs EQ b~bukrs
     AND a~belnr EQ b~belnr
     AND a~gjahr EQ b~gjahr
    LEFT OUTER JOIN t003t AS c
      ON a~blart EQ c~blart
     AND c~spras EQ sy-langu
    LEFT OUTER JOIN tbslt AS d
      ON b~bschl EQ d~bschl
     AND b~umskz EQ d~umskz
     AND d~spras EQ sy-langu
   WHERE a~bukrs IN so_burks
     AND a~gjahr EQ pa_gjahr.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM make_fcat USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE gs_fcat-ref_table.
    WHEN 'FCSL'.
      gs_fcat-cfieldname = 'RFCCUR'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_list .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
      is_variant      = gs_variant
    CHANGING
      it_outtab       = gt_data
      it_fieldcatalog = gt_fcat.

ENDFORM.

FORM change_data .

  DATA : lv_tabix TYPE sy-tabix.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    CASE gs_data-shkzg.

      WHEN 'H'.
        gs_data-fcsl *= -1.

    ENDCASE.

    gs_data-rfccur = 'KRW'.
    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING fcsl rfccur.

  ENDLOOP.

  CLEAR lv_tabix.

ENDFORM.
