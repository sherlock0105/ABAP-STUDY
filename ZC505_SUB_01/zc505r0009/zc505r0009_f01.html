FORM get_data .

  REFRESH : gt_data.

  SELECT a~matnr a~mtart a~meins
         b~werks b~pstat b~ekgrp b~dispo
         c~maktx
         d~mtbez
         e~wgbez
         f~eknam
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM mara AS a
   INNER JOIN marc AS b
      ON a~matnr EQ b~matnr
    LEFT OUTER JOIN makt AS c
      ON a~matnr EQ c~matnr
     AND c~spras EQ sy-langu
    LEFT OUTER JOIN t134t AS d
      ON a~mtart EQ d~mtart
     AND d~spras EQ sy-langu
    LEFT OUTER JOIN t023t AS e
      ON a~matkl EQ e~matkl
     AND e~spras EQ sy-langu
    LEFT OUTER JOIN t024 AS f
      ON b~ekgrp EQ f~ekgrp
   WHERE a~matnr IN so_matnr
     AND a~mtart IN so_mtart
     AND b~werks IN so_werks.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ENDIF.

ENDFORM.

FORM display_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_data[]
      it_fieldcatalog = gt_fcat.

ENDFORM.
