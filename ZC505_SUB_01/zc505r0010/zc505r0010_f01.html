FORM get_air_info .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~carrid AS air_code a~carrname AS air_name
         b~connid b~airpfrom b~deptime
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM scarr AS a
   INNER JOIN spfli AS b
      ON a~carrid EQ b~carrid
   WHERE a~carrid IN so_carr.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM make_fcat .

*carrid 필드는 key 로 지정
*ITAB 의 AIR_CODE 라는 필드는 SCARR-CARRID 필드를 참조
*CARRID 필드의 DE 와 DO 의 속성을 적용하겠다.
  gs_fcat-key       = 'X'.
  gs_fcat-fieldname = 'AIR_CODE'. "ITAB 의 필드명을 ALV 에 등록
  gs_fcat-ref_table = 'SCARR'.
  gs_fcat-ref_field = 'CARRID'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

  gs_fcat-key       = 'X'.
  gs_fcat-fieldname = 'CONNID'.
  gs_fcat-ref_table = 'SPFLI'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

  gs_fcat-fieldname = 'AIR_NAME'.
  gs_fcat-ref_table = 'SCARR'.
  gs_fcat-ref_field = 'CARRNAME'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

  gs_fcat-fieldname = 'AIRPFROM'.
  gs_fcat-ref_table = 'SPFLI'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

  gs_fcat-fieldname = 'DEPTIME'.
  gs_fcat-ref_table = 'SPFLI'.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM create_alv .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_layout       = gs_layout
      i_save          = 'A'
      i_default       = 'X'
    CHANGING
      it_fieldcatalog = gt_fcat
      it_outtab       = gt_data.

ENDFORM.
