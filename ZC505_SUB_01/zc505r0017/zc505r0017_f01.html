FORM get_data .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~aufnr a~gstrp a~gltrp a~plnbez a~gamng a~gmein
         b~posnr b~matnr b~psmng b~amein b~bwtar
         c~maktx AS autxt
         d~maktx AS matxt
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM afko AS a
   INNER JOIN afpo AS b
      ON a~aufnr EQ b~aufnr
    LEFT OUTER JOIN makt AS c
      ON a~plnbez EQ c~matnr
     AND c~spras EQ sy-langu
    LEFT OUTER JOIN makt AS d
      ON b~matnr EQ d~matnr
     AND d~spras EQ sy-langu
   WHERE a~plnbez IN so_pln
     AND b~pwerk EQ pa_pwerk.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM move_data .

  REFRESH : gt_data2.

  DATA : lv_tabix TYPE sy-tabix.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    CASE gs_data-bwtar.

      WHEN '02'.
        MOVE-CORRESPONDING gs_data TO gs_data2.
        APPEND gs_data2 TO gt_data2.
        CLEAR  gs_data2.

    ENDCASE.

    CLEAR : lv_tabix.

  ENDLOOP.

ENDFORM.

FORM make_fcat USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.

    WHEN 'PSMNG'.
      gs_fcat-qfieldname = 'AMEIN'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR gs_fcat.
ENDFORM.

FORM display_list .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
    CHANGING
      it_outtab       = gt_data2
      it_fieldcatalog = gt_fcat.

ENDFORM.
