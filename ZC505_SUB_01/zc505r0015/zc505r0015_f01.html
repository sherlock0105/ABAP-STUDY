FORM get_data .

  CLEAR   : gs_data.
  REFRESH : gt_data.

  SELECT a~matnr a~mtart a~matkl a~mbrsh
         b~werks
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM mara AS a
   INNER JOIN marc AS b
      ON a~matnr EQ b~matnr
   WHERE b~werks EQ pa_werks
     AND a~matnr IN so_matnr.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM get_text .

  CLEAR   : gs_t134t, gs_t023t, gs_t137t.
  REFRESH : gt_t134t, gt_t023t, gt_t137t.

  SELECT mtart mtbez
    INTO CORRESPONDING FIELDS OF TABLE gt_t134t
    FROM t134t
   WHERE spras EQ sy-langu.

  SELECT matkl wgbez
    INTO CORRESPONDING FIELDS OF TABLE gt_t023t
    FROM t023t
   WHERE spras EQ sy-langu.

  SELECT mbrsh mbbez
    INTO CORRESPONDING FIELDS OF TABLE gt_t137t
    FROM t137t
   WHERE spras EQ sy-langu.

ENDFORM.

FORM data_loop .

  LOOP AT gt_data INTO gs_data.

    gv_tabix = sy-tabix.

    READ TABLE gt_t134t INTO gs_t134t WITH KEY mtart = gs_data-mtart.

    IF sy-subrc EQ 0.
      gs_data-mtbez = gs_t134t-mtbez.

      MODIFY gt_data FROM gs_data INDEX gv_tabix
      TRANSPORTING mtbez.

      CLEAR gs_t134t.

    ENDIF.

    READ TABLE gt_t023t INTO gs_t023t WITH KEY matkl = gs_data-matkl.

    IF sy-subrc EQ 0.
      gs_data-wgbez = gs_t023t-wgbez.

      MODIFY gt_data FROM gs_data INDEX gv_tabix
      TRANSPORTING wgbez.

      CLEAR gs_t023t.

    ENDIF.

    READ TABLE gt_t137t INTO gs_t137t WITH KEY mbrsh = gs_data-mbrsh.

    IF sy-subrc EQ 0.
      gs_data-mbbez = gs_t137t-mbbez.

      MODIFY gt_data FROM gs_data INDEX gv_tabix
      TRANSPORTING mbbez.

      CLEAR gs_t137t.

    ENDIF.

    CLEAR : gv_tabix.

  ENDLOOP.

ENDFORM.

FORM set_fcat_layout .

  gs_layout-zebra      = 'X'.
  gs_layout-cwidth_opt = 'A'.
  gs_layout-sel_mode   = 'D'.

  gs_stable-col = 'X'.
  gs_stable-row = 'X'.

  PERFORM make_fcat USING :
      'X'  'MATNR' ' '  'MARA'   ' ',
      'X'  'WERKS' ' '  'MARC'   ' ',
      ' '  'MTART' ' '  'MARA'   ' ',
      ' '  'MTBEZ' ' '  'T134T'  ' ',
      ' '  'MATKL' ' '  'MARA'   ' ',
      ' '  'WGBEZ' ' '  'T023T'  ' ',
      ' '  'MBRSH' ' '  'MARA'   ' ',
      ' '  'MBBEZ' ' '  'T137T'  ' '.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*& -->  p1        text
*& <--  p2        text
*&---------------------------------------------------------------------*
FORM make_fcat USING pv_key pv_field pv_text pv_ref_table pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_list .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      i_save          = 'A'
      i_default       = 'X'
      is_layout       = gs_layout
      is_variant      = gs_variant
    CHANGING
      it_outtab       = gt_data
      it_fieldcatalog = gt_fcat.

ENDFORM.
