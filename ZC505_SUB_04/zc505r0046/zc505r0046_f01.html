FORM get_mat .

  _clear gs_mat gt_mat.

  SELECT matnr werks ekgrp
         dispo beskz
    INTO CORRESPONDING FIELDS OF TABLE gt_mat
    FROM zc505t0042
   WHERE werks EQ pa_werks
     AND matnr IN so_matnr
     AND ekgrp IN so_ekgrp.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant
      i_save               = 'A'
      i_default            = 'X'
      is_layout            = gs_layout
      it_toolbar_excluding = gt_exclude
    CHANGING
      it_outtab            = gt_mat
      it_fieldcatalog      = gt_fcat.

ENDFORM.

FORM set_style .
  "db 에서 가져온 데이터 pk 는 수정 못하게
  "이거 조금 헷갈림

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  LOOP AT gt_mat INTO gs_mat.

    lv_tabix = sy-tabix.

    REFRESH gs_mat-style.

    ls_style-fieldname = 'MATNR'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_mat-style.
    CLEAR  ls_style.

    MODIFY gt_mat FROM gs_mat INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH 'First, select rows' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row. " 선택한 행을 읽기

    READ TABLE gt_mat INTO gs_mat INDEX gs_row-index.

    IF sy-subrc EQ 0.
      APPEND gs_mat TO gt_mat_del.
      CLEAR  gs_mat.

      DELETE gt_mat INDEX gs_row-index.
    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.
