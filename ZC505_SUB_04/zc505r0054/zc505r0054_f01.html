FORM get_data .

  _clear gs_data gt_data.
  REFRESH gt_data_del.

  SELECT a~werks a~mtart a~flref a~envop
         a~kkref a~vmtpo a~menge a~meins
         a~erdat a~erzet a~ernam
         a~aedat a~aezet a~aenam
         b~mtbez
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM ZC505t0044 AS a
    LEFT OUTER JOIN t134t AS b
      ON a~mtart EQ b~mtart
     AND b~spras EQ sy-langu
   WHERE werks   EQ pa_werks
   ORDER BY erdat erzet.

ENDFORM.

FORM display_screen .

    EXPORTING

      extension = 5000.                           " Control Extension


  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.  " Parent Container

  gs_variant-report = sy-repid.

  SET HANDLER lcl_event_handler=>handle_changed_finished FOR gcl_grid.

  CALL METHOD gcl_grid->register_edit_event
    EXPORTING
      i_event_id = gcl_grid->mc_evt_modified.                  " Event ID

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant                 " Layout
      i_save               = 'A'                      " Save Layout
      i_default            = 'X'                     " Default Display Variant
      is_layout            = gs_layout                 " Layout
      it_toolbar_excluding = gt_exclude               " Excluded Toolbar Standard Functions
    CHANGING
      it_outtab            = gt_data                " Output Table
      it_fieldcatalog      = gt_fcat.               " Field Catalog


ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

*  REFRESH : gt_data_del. "????

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.                 " Indexes of Selected Rows

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH 'First, Select rows' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_data INTO gs_data INDEX gs_row-index.

FORM save_data .

  DATA : ls_save   TYPE ZC505t0044,
         lt_save   LIKE TABLE OF ls_save,
         lt_del    LIKE TABLE OF ls_save,
         lv_commit.

  _clear ls_save lt_save. " 이거 까먹
  CLEAR : lv_commit. " 이거 까먹음.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_data_del IS NOT INITIAL.

    MOVE-CORRESPONDING gt_data_del TO lt_del.

    DELETE ZC505t0044 FROM TABLE lt_del.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  LOOP AT gt_data INTO gs_data WHERE mark EQ 'X'.

    MOVE-CORRESPONDING gs_data TO ls_save.

    CHECK gs_data-mtart IS NOT INITIAL.

    IF ls_save-erdat IS INITIAL. "이거 못외움

      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.

    ELSE.

      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.
    MODIFY ZC505t0044 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.
  ENDIF.

  IF lv_commit IS NOT INITIAL.
    COMMIT WORK AND WAIT.

    REFRESH gt_data_del. " 이거 까먹음
    PERFORM clear_mark.
    MESSAGE s002.

  ELSE.
    MESSAGE s000 WITH 'Data not found' DISPLAY LIKE 'E'.
  ENDIF.

ENDFORM.

FORM handle_data_changed  USING  pcl_changed TYPE REF TO cl_alv_changed_data_protocol.

  DATA : ls_modi  TYPE lvc_s_modi,
         lv_mtbez TYPE t134t-mtbez.

  LOOP AT pcl_changed->mt_mod_cells INTO ls_modi.

    READ TABLE gt_data INTO gs_data INDEX ls_modi-row_id.

    IF sy-subrc NE 0.
      CONTINUE. " LOOP 로 돌아감
    ENDIF.

    CASE ls_modi-fieldname.
      WHEN 'MTART'.

        SELECT SINGLE mtbez
          INTO lv_mtbez
          FROM t134t
         WHERE mtart = ls_modi-value
           AND spras EQ sy-langu.

        CALL METHOD pcl_changed->modify_cell
          EXPORTING
            i_row_id    = ls_modi-row_id          " Row ID
            i_fieldname = 'MTBEZ'                 " Field Name
            i_value     = lv_mtbez.               " Value

    ENDCASE.

    gs_data-mark = 'X'.

    MODIFY gt_data FROM gs_data INDEX ls_modi-row_id
    TRANSPORTING mark.


  ENDLOOP.

ENDFORM.

FORM handle_changed_finished  USING    pv_modified
                                       pt_modi TYPE lvc_t_modi.

  DATA : ls_modi  TYPE lvc_s_modi.
*  lv_mtbez TYPE t134t-mtbez.
*
*  IF pv_modified IS INITIAL.
*    EXIT.
*  ENDIF.

  CHECK pv_modified IS NOT INITIAL. "밑에 수행하려고 체크

  LOOP AT pt_modi INTO ls_modi.

    READ TABLE gt_data INTO gs_data INDEX ls_modi-row_id.

    IF sy-subrc NE 0.
      CONTINUE.
    ENDIF.

    gs_data-mark = 'X'.

    CASE ls_modi-fieldname.
      WHEN 'MTART'.

        SELECT SINGLE mtbez
          INTO gs_data-mtbez "인터널 테이블이 끝난 상태라 가능하다.
          FROM t134t
         WHERE mtart = ls_modi-value
           AND spras = sy-langu.

    ENDCASE.

    MODIFY gt_data FROM gs_data INDEX ls_modi-row_id
    TRANSPORTING mark mtbez.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.
