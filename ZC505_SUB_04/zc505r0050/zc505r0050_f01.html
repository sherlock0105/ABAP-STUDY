FORM get_class .

  _clear gs_class gt_class.

  SELECT class cl_num name phone gender
         erdat  erzet ernam aedat aezet aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_class
    FROM zc505t0002
   WHERE class EQ pa_class

FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  LOOP AT gt_class INTO gs_class.

    lv_tabix = sy-tabix.

    REFRESH gs_class-style.

    ls_style-fieldname = 'CL_NUM'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_class-style.
    CLEAR ls_style.

    MODIFY gt_class FROM gs_class INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

ENDFORM.

  CALL METHOD gcl_grid->set_table_for_first_display

      it_toolbar_excluding = gt_exclude             " Excluded Toolbar Standard Functions

      it_outtab            = gt_class                " Output Table
      it_fieldcatalog      = gt_fcat.                " Field Catalog

FORM save_data .

  DATA : ls_save   TYPE zc505t0002,
         lt_save   TYPE TABLE OF zc505t0002,
         lv_commit.

  _clear ls_save lt_save.
  CLEAR lv_commit.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_class_del IS NOT INITIAL.

    DELETE zc505t0002 FROM TABLE gt_class_del.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  LOOP AT gt_class INTO gs_class WHERE mark EQ 'X'.

    MOVE-CORRESPONDING gs_class TO ls_save.

    IF ls_save-erdat IS INITIAL.

      ls_save-erdat = sy-datum. "시스템 현재 날짜
*      ls_save-erdat = sy-datlo. "현지 날짜
      ls_save-erzet = sy-uzeit. "시스템 현재시간
*      ls_save-erzet = sy-timlo. "현지 시간
      ls_save-ernam = sy-uname.

    ELSE.
      ls_save-aedat = sy-datum. "시스템 현재 날짜
*      ls_save-aedat = sy-datlo. "현지 날짜
      ls_save-aezet = sy-uzeit. "시스템 현재시간
*      ls_save-aezet = sy-timlo. "현지 시간
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.
    MODIFY zc505t0002 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.
    COMMIT WORK AND WAIT.
    REFRESH gt_class_del. "del 를 했으니까 클리어 해준다. 또 해줄 수 있음.

    PERFORM clear_mark.

    MESSAGE s002.
  ELSE.
    MESSAGE s000 WITH 'Data was not changed' DISPLAY LIKE 'W'.
  ENDIF.

  LOOP AT gt_class INTO gs_class WHERE mark EQ 'X'.
