FORM set_style .
  "db 에서 가져온 데이터 pk 는 수정 못하게

         ls_style TYPE lvc_s_styl.

    lv_tabix = sy-tabix.

    REFRESH gs_idol-style.

    ls_style-fieldname = 'GROUP_ID'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_idol-style.
    CLEAR  ls_style.

    ls_style-fieldname = 'MEMBER_ID'.

    APPEND ls_style TO gs_idol-style.
    CLEAR  ls_style.

    MODIFY gt_idol FROM gs_idol INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant
      i_save               = 'A'
      i_default            = 'X'
      is_layout            = gs_layout
      it_toolbar_excluding = gt_exclude
    CHANGING
      it_outtab            = gt_idol
      it_fieldcatalog      = gt_fcat.

ENDFORM.
