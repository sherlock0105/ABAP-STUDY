FORM get_data .

  SELECT matnr werks ekgrp dispo beskz
         erdat erzet ernam

FORM delete_row .
  DATA lv_cnt TYPE i.


  _clear gs_row gt_row.
  CLEAR lv_cnt.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.                 " Indexes of Selected Rows

  lv_cnt = lines( gt_row ). "행의 개수 파악

  IF lv_cnt < 1.
    MESSAGE s000 WITH 'First, select rows' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_data INTO gs_data INDEX gs_row-index.

    IF sy-subrc EQ 0.

      APPEND gs_data TO gt_data_del. "여기 헷갈림
      CLEAR gs_data.

      DELETE gt_data INDEX gs_row-index. "여기 헷갈림
    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM save_data .

  DATA : ls_save   TYPE zc505t0042,
         lt_save   LIKE TABLE OF ls_save,
         lt_del    LIKE TABLE OF ls_save,
         lv_commit.

  _clear ls_save lt_save.
  CLEAR lv_commit.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_data_del IS NOT INITIAL.

    MOVE-CORRESPONDING gt_data_del TO lt_del.

    DELETE zc505t0042 FROM TABLE lt_del.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  LOOP AT gt_data INTO gs_data WHERE mark EQ 'X'.

    MOVE-CORRESPONDING gs_data TO ls_save.

    IF ls_save-erdat IS INITIAL.

      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.

    ELSE.

      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.
    MODIFY zc505t0042 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.
    COMMIT WORK AND WAIT.
    REFRESH gt_data_del. " 이거 이해 안돼

    PERFORM clear_mark. "지워주고

    MESSAGE s002.

  ELSE.

    MESSAGE s000 WITH 'Data was not changed' DISPLAY LIKE 'W'.

  ENDIF.


ENDFORM.

*& Form handle_data_changed
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> ER_DATA_CHANGED
*&---------------------------------------------------------------------*
FORM handle_data_changed  USING    pcl_changed TYPE REF TO cl_alv_changed_data_protocol.

  DATA : ls_modi TYPE lvc_s_modi.

  LOOP AT pcl_changed->mt_mod_cells INTO ls_modi.

    READ TABLE gt_data INTO gs_data INDEX ls_modi-row_id.

    IF sy-subrc NE 0.
      CONTINUE.
    ENDIF.

    gs_data-mark = 'X'.

    MODIFY gt_data FROM gs_data INDEX ls_modi-row_id
    TRANSPORTING mark.

  ENDLOOP.

ENDFORM.

FORM check_param .

  DATA : lv_werks TYPE zc505t0042-werks.

  IF pa_werks IS INITIAL.
    MESSAGE s000 WITH 'Input data' DISPLAY LIKE 'E'.
    STOP.
  ENDIF.

  SELECT SINGLE werks
    INTO lv_werks
    FROM t001w
    WHERE werks EQ pa_werks.

  IF sy-subrc NE 0.
    MESSAGE s000 WITH 'Invalid value' DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

ENDFORM.
