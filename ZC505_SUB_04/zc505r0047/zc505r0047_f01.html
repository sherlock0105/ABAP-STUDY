FORM get_baseball_data .

  _clear gs_data gt_data.

  SELECT team team_name area win
         erdat erzet ernam
         aedat aezet aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM zc505t0012
   WHERE team IN so_team.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left
      extension = 5000.

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant
      i_save               = 'A'
      i_default            = 'X'
      is_layout            = gs_layout
      it_toolbar_excluding = gt_exclude
    CHANGING
      it_outtab            = gt_data
      it_fieldcatalog      = gt_fcat.

FORM save_data .

  DATA : ls_save TYPE zc505t0012,
         lt_save LIKE TABLE OF zc505t0012,
         lt_del  LIKE TABLE OF ls_save.

  CALL METHOD : gcl_grid->check_changed_data.

  IF gt_data_del IS NOT INITIAL. "이게 비어있지 않으면

    MOVE-CORRESPONDING gt_data_del TO lt_del.

    DELETE zc505t0012 FROM TABLE lt_del.

  ENDIF.

  LOOP AT gt_data INTO gs_data.

    IF gs_data-team IS NOT INITIAL.

      MOVE-CORRESPONDING gs_data TO ls_save.

      IF ls_save-erdat IS INITIAL.

        ls_save-erdat = sy-datum.
        ls_save-erzet = sy-uzeit.
        ls_save-ernam = sy-uname.

      ELSE.
        ls_save-aedat = sy-datum.
        ls_save-aezet = sy-uzeit.
        ls_save-aenam = sy-uname.

      ENDIF.

      APPEND ls_save TO lt_save.
      CLEAR  ls_save.

    ENDIF.

  ENDLOOP.


  IF lt_save IS NOT INITIAL.

    MODIFY zc505t0012 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      COMMIT WORK AND WAIT.

      REFRESH gt_data_del.
      MESSAGE s002.
    ENDIF.

  ENDIF.

ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows   "기억이 안나ㅠ
    IMPORTING
      et_index_rows = gt_row.

  IF gt_row IS INITIAL. " gt_row 값이 초기값이면

    MESSAGE s000 WITH 'First, select rows' DISPLAY LIKE 'E'. "message
    EXIT.
  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_data INTO gs_data INDEX gs_row-index.

    IF sy-subrc EQ 0.
      APPEND gs_data TO gt_data_del.
      CLEAR  gs_data.

      DELETE gt_data INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.
