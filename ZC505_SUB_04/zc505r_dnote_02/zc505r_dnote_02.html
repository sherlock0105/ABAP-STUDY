FORM updata_data .
*gt_class 에 있는 phone 을 '11111111' 로 바꿔보자
  DATA : ls_update TYPE zc505t0002,
         lt_update TYPE TABLE OF zc505t0002,
         lv_tabix  TYPE sy-tabix.

  _clear ls_update lt_update.

  LOOP AT gt_class INTO gs_class.

    lv_tabix = sy-tabix.

    gs_class-phone = '11111111'.

    MODIFY gt_class FROM gs_class INDEX lv_tabix
    TRANSPORTING phone.

  ENDLOOP.

  MOVE-CORRESPONDING gt_class TO lt_update.

  IF lt_update IS NOT INITIAL.

FORM delete_data .

  DATA : ls_del TYPE zc505t0002,
         lt_del TYPE TABLE OF zc505t0002.

  _clear ls_del lt_del.

* gender 가 'F' 인 레코드를 삭제해보자

  LOOP AT gt_class INTO gs_class WHERE gender EQ 'F'.

    MOVE-CORRESPONDING gs_class TO ls_del.

    APPEND ls_del TO lt_del.
    CLEAR  ls_del.

  ENDLOOP.

  IF lt_del IS NOT INITIAL.

    DELETE zc505t0002 FROM TABLE lt_del.
