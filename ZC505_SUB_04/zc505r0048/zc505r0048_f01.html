FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    REFRESH gs_data-style. "루프 돌때마다 빈값으로 만들어주기

    CLEAR ls_style.

    ls_style-fieldname = 'CAPITAL'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_data-style.
    CLEAR  ls_style.

    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING style.

  ENDLOOP.

ENDFORM.

FORM save_data .

  DATA : ls_save TYPE zc505t0007,
         lt_save LIKE TABLE OF ls_save,
         lt_del  LIKE TABLE OF ls_save.

  CALL METHOD : gcl_grid->check_changed_data.

*delete
  IF gt_data_del IS NOT INITIAL.

    MOVE-CORRESPONDING gt_data_del TO lt_del.
    DELETE zc505t0007 FROM TABLE lt_del.

  ENDIF.

  LOOP AT gt_data INTO gs_data.

    IF gs_data-capital IS NOT INITIAL.

      MOVE-CORRESPONDING gs_data TO ls_save.

      IF ls_save-erdat IS INITIAL. "이게 초기값이면

        ls_save-erdat = sy-datum.
        ls_save-erzet = sy-uzeit.
        ls_save-ernam = sy-uname.

      ELSE.
        ls_save-aedat = sy-datum.
        ls_save-aezet = sy-uzeit.
        ls_save-aenam = sy-uname.

      ENDIF.

      APPEND ls_save TO lt_save.
      CLEAR  ls_save.

    ENDIF.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.
    MODIFY zc505t0007 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      COMMIT WORK AND WAIT.

      REFRESH gt_data_del.
      MESSAGE s002.
    ENDIF.

  ENDIF.

ENDFORM.

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.             " Indexes of Selected Rows

"gt_row 가 선택하지 않으면 메세지 + exit.
  IF gt_row IS INITIAL.
    MESSAGE s000 WITH 'First, select rows' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.
"지우기 전에 sort 로 정리
  SORT gt_row BY index DESCENDING.
"loop 돌리기 선택한 row를 찾아
  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_data INTO gs_data INDEX gs_row-index.

    IF sy-subrc EQ 0. "헷갈
      APPEND gs_data TO gt_data_del.
      CLEAR  gs_data.

      DELETE gt_data INDEX gs_row-index.
    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.
