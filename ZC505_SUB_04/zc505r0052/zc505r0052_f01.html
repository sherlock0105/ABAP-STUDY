  IF pa_kokrs IS INITIAL.
    MESSAGE s000 WITH 'input data' DISPLAY LIKE 'E'.
    STOP.
  ENDIF.

    INTO lv_korks
    FROM tka01
    WHERE kokrs EQ pa_kokrs.

FORM get_data .

  _clear gs_data gt_data.
  REFRESH gt_data_del.

  SELECT a~kokrs a~kostl a~bukrs a~kosar a~verak
         a~wrbtr a~waers a~bkzer
         a~erdat a~erzet a~ernam
         a~aedat a~aezet a~aenam
         b~ltext
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM zc505t0043 AS a
    LEFT OUTER JOIN cskt AS b
      ON  a~kokrs EQ b~kokrs
     AND a~kostl EQ b~kostl
     AND b~spras EQ sy-langu
   WHERE a~kokrs EQ pa_kokrs
   ORDER BY a~kokrs.

ENDFORM.

      it_fieldcatalog      = gt_fcat.               " Field Catalog

FORM delete_row .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.            " Indexes of Selected Rows

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH 'First, select rows' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  SORT gt_row BY index DESCENDING.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_data INTO gs_data INDEX gs_row-index.

    IF sy-subrc EQ 0.

      APPEND gs_data TO  gt_data_del.
      CLEAR gs_data.

      DELETE gt_data INDEX gs_row-index.

    ENDIF.

  ENDLOOP.

  PERFORM refresh_grid.

ENDFORM.

FORM set_style .

  DATA : lv_tablx TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  LOOP AT gt_data INTO gs_data.

    lv_tablx = sy-tabix.

    REFRESH gs_data-style.

    ls_style-fieldname = 'KOKRS'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_disabled.

    APPEND ls_style TO gs_data-style.
    CLEAR  ls_style.

    TRANSPORTING style.

  ENDLOOP.

FORM save_data .

  DATA : ls_save   TYPE zc505t0043,
         lt_save   LIKE TABLE OF ls_save,
         lt_del    LIKE TABLE OF ls_save,
         lv_commit.

  _clear ls_save lt_save.
  CLEAR lv_commit.

  CALL METHOD : gcl_grid->check_changed_data.

  IF gt_data_del IS NOT INITIAL.
    MOVE-CORRESPONDING gt_data_del TO lt_del.

    DELETE zc505t0043 FROM TABLE lt_del.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.
  ENDIF.

  LOOP AT gt_data INTO gs_data WHERE mark EQ 'X'.

    MOVE-CORRESPONDING gs_data TO ls_save.

    CHECK gs_data-kostl IS NOT INITIAL.

    IF ls_save-erdat IS INITIAL.

      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.

    ELSE.

      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.

    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.
    MODIFY zc505t0043 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.

    COMMIT WORK AND WAIT.

    REFRESH gt_data_del.

    PERFORM clear_mark.
    MESSAGE s002.

  ELSE.

    MESSAGE s000 WITH 'Data was not changed' DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form handle_data_changed
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> ER_DATA_CHANGED_>MT_MOD_CELLS
*&---------------------------------------------------------------------*
FORM handle_data_changed  USING  pcl_changed TYPE REF TO cl_alv_changed_data_protocol.

  DATA : ls_modi  TYPE lvc_s_modi,
         lv_ltext TYPE cskt-ltext.

  LOOP AT pcl_changed->mt_mod_cells INTO ls_modi.

    READ TABLE gt_data INTO gs_data INDEX ls_modi-row_id.

    IF sy-subrc NE 0.
      CONTINUE.
    ENDIF.

    gs_data-mark = 'X'.

    CASE ls_modi-fieldname.
      WHEN 'KOSTL'.

        SELECT SINGLE ltext
         INTO lv_ltext
         FROM cskt
        WHERE spras EQ sy-langu
          AND kokrs EQ pa_kokrs
          AND kostl EQ ls_modi-value.

        CALL METHOD pcl_changed->modify_cell
          EXPORTING
            i_row_id    = ls_modi-row_id    " Row ID
            i_fieldname = 'LTEXT'           " Field Name
            i_value     = lv_ltext.         " Value

    ENDCASE.

    MODIFY gt_data FROM gs_data INDEX ls_modi-row_id
    TRANSPORTING mark.

  ENDLOOP.

ENDFORM.
