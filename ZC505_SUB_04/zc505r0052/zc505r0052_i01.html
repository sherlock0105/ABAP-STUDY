MODULE user_command_0100 INPUT.


  CASE gv_okcode.

    WHEN 'REFRESH'.              "조회/새로고침 버튼
      CLEAR gv_okcode.           "OKCODE 재사용 방지

      PERFORM get_data.          "DB 재조회
      PERFORM set_style.         "ALV 스타일/편집속성 재설정
      PERFORM refresh_grid.      "ALV 화면 갱신

    WHEN 'CREATE'.               "행 추가 버튼
      CLEAR gv_okcode.

      PERFORM create_row.        "신규 행 추가(내부테이블)
      PERFORM refresh_grid.      "ALV 즉시 반영

    WHEN 'DELETE'.               "삭제 버튼
      CLEAR gv_okcode.

      PERFORM delete_row.        "선택 행 삭제 처리

    WHEN 'SAVE'.                 "저장 버튼
      CLEAR gv_okcode.

      PERFORM save_data.         "변경 데이터 DB 반영
      PERFORM get_data.          "DB 기준 재조회
      PERFORM set_style.         "편집/색상/스타일 재적용
      PERFORM refresh_grid.      "ALV 재표시

WHEN 'TOGGLE'.
  CLEAR gv_okcode.

  PERFORM toggle_display.

  ENDCASE.

ENDMODULE.
