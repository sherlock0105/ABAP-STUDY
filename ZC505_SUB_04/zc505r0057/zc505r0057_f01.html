FORM check_param .

  DATA : lv_bukrs TYPE bkpf-bukrs.

  IF pa_burks IS INITIAL OR pa_gjahr IS INITIAL.

    MESSAGE s000 WITH 'Input data' DISPLAY LIKE 'E'.
    STOP.

  ENDIF.

  SELECT SINGLE bukrs
    INTO lv_bukrs
    FROM t001
   WHERE bukrs EQ pa_burks.

  IF sy-subrc NE 0.
    MESSAGE s000 WITH 'Invalid data' DISPLAY LIKE 'E'.
    STOP.
  ENDIF.

ENDFORM.

FORM get_data .

  _clear gs_data gt_data.

  SELECT a~blart a~budat "bkpf
         b~belnr b~buzei b~bschl b~koart
         b~gsber b~hkont b~shkzg b~dmbtr b~pswsl
         c~ltext
         d~gtext
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM bkpf AS a
   INNER JOIN bseg AS b
      ON a~bukrs EQ b~bukrs
     AND a~belnr EQ b~belnr
     AND a~gjahr EQ b~gjahr
    LEFT OUTER JOIN t003t AS c
      ON a~blart EQ c~blart
     AND c~spras EQ sy-langu
    LEFT OUTER JOIN tgsbt AS d
      ON b~gsber EQ d~gsber
     AND d~spras EQ sy-langu
   WHERE a~bukrs EQ pa_burks
     AND a~gjahr EQ pa_gjahr.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left     " Side to Which Control is Docked
      extension = 5000.                           " Control Extension

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.                 " Parent Container

  IF gcl_handler IS NOT BOUND.

    CREATE OBJECT gcl_handler.

  ENDIF.

  SET HANDLER : gcl_handler->handle_on_button_click FOR gcl_grid,
                gcl_handler->handle_double_click     FOR gcl_grid,
                gcl_handler->handle_changed_finished FOR gcl_grid,
                gcl_handler->handle_toolbar          FOR gcl_grid,
                gcl_handler->handle_user_command     FOR gcl_grid.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant           = gs_variant               " Layout
      i_save               = 'A'                      " Save Layout
      i_default            = 'X'                      " Default Display Variant
      is_layout            = gs_layout                " Layout
      it_toolbar_excluding = gt_exclude               " Excluded Toolbar Standard Functions
    CHANGING
      it_outtab            = gt_data                  " Output Table
      it_fieldcatalog      = gt_fcat.                 " Field Catalog


ENDFORM.

FORM set_style .

  DATA : lv_tabix TYPE sy-tabix,
         ls_style TYPE lvc_s_styl.

  LOOP AT gt_data INTO gs_data.

    lv_tabix = sy-tabix.

    REFRESH gs_data-style.

    CHECK gs_data-buzei EQ '0001'.

    ls_style-fieldname = 'BELNR'.
    ls_style-style     = cl_gui_alv_grid=>mc_style_button.

    INSERT ls_style INTO TABLE gs_data-style.
    CLEAR  ls_style.

    MODIFY gt_data FROM gs_data INDEX lv_tabix
    TRANSPORTING style.

    CASE gs_data-buzei.
      WHEN '0001'.
        ls_style-fieldname = 'BELNR'.
        ls_style-style     = cl_gui_alv_grid=>mc_style_button.

        INSERT ls_style INTO TABLE gs_data-style.
        CLEAR  ls_style.

        MODIFY gt_data FROM gs_data INDEX lv_tabix
        TRANSPORTING style.

    ENDCASE.

  ENDLOOP.

ENDFORM.
