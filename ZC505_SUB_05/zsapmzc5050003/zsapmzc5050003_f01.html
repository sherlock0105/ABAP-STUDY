*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING   pv_key
                        pv_field
                        pv_text
                        pv_ref_table
                        pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.
    WHEN 'LABST'.
      gs_fcat-qfieldname = 'MEINS'.
    WHEN 'NETPR'.
      gs_fcat-cfieldname = 'WAERS'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM save_data .

  DATA : ls_save TYPE zc505t0046. "db용 만들기, 테이블 타입을 정확하게 맞춘다.

  IF gs_data-goods IS INITIAL.
    MESSAGE s000 WITH 'Input goods data' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  MOVE-CORRESPONDING gs_data TO ls_save.

  IF ls_save-erdat IS INITIAL.

    ls_save-erdat = sy-datum.
    ls_save-erzet = sy-uzeit.
    ls_save-ernam = sy-uname.

  ELSE.

    ls_save-aedat = sy-datum.
    ls_save-aezet = sy-uzeit.
    ls_save-aenam = sy-uname.

  ENDIF.

  MODIFY zc505t0046 FROM ls_save.

  IF sy-dbcnt > 0.
    COMMIT WORK AND WAIT.

    PERFORM get_data.
    PERFORM refresh_grid.
    MESSAGE s002.

  ELSE.
    ROLLBACK WORK. "이거 왜함?
    MESSAGE s004 DISPLAY LIKE 'E'.
  ENDIF.

ENDFORM.

FORM get_data .

  CLEAR   gs_data.
  REFRESH gt_data.

  SELECT goods category goods_count
         goods_desc labst meins netpr waers
         erdat erzet ernam aedat aezet aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM zc505t0046
   ORDER BY goods.

ENDFORM.

FORM delete_data .

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.          " Indexes of Selected Rows

*  IF gt_row IS INITIAL.
*    EXIT.
*  ENDIF.

  LOOP AT gt_row INTO gs_row.

    READ TABLE gt_data INTO gs_data INDEX gs_row-index.

    DELETE gt_data INDEX gs_row-index.

  ENDLOOP.


  IF gs_data-goods IS INITIAL.

    MESSAGE s000 WITH 'Goods is required field'
    DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  DELETE FROM zc505t0046
  WHERE goods = gs_data-goods.

  IF sy-subrc EQ 0.
    COMMIT WORK AND WAIT.
    MESSAGE s003.
    PERFORM get_data.
    PERFORM refresh_grid.
  ELSE.
    ROLLBACK WORK.
    MESSAGE s005 DISPLAY LIKE 'E'.
  ENDIF.

ENDFORM.

FORM f4_goods .

  DATA : BEGIN OF ls_goods,
           goods      TYPE zc505t0014-goods,
           goods_desc TYPE zc505t0014-goods_desc,
         END OF ls_goods,
         lt_goods LIKE TABLE OF ls_goods.

  SELECT goods goods_desc
    INTO CORRESPONDING FIELDS OF TABLE lt_goods
    FROM zc505t0014.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'
    EXPORTING
      retfield     = 'GOODS'
      dynpprog     = sy-repid
      dynpnr       = sy-dynnr
      dynprofield  = 'GS_DATA-GOODS'
      window_title = 'Goods Master'
      value_org    = 'S' " S 로 해야 더블클릭으로 동작함
    TABLES
      value_tab    = lt_goods.

ENDFORM.
