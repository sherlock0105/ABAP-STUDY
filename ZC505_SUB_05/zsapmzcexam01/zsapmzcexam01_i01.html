MODULE exit_0100 INPUT.

  CALL METHOD : gcl_grid->free, gcl_container->free.   "생성된 grid 와 container 를 메모리에서 해제

  FREE: gcl_grid, gcl_container. "객체 변수 초기화

  CASE gv_okcode.

    WHEN 'EXIT'.          "Exit 면 프로그램 종료
      LEAVE PROGRAM.

    WHEN OTHERS.          "현재 화면을 닫고 이전으로 이동
      LEAVE TO SCREEN 0.

  ENDCASE.

ENDMODULE.

MODULE user_command_0100 INPUT.

  CASE gv_okcode.

    WHEN 'REFRESH'.             "조회/새로고침 버튼
      CLEAR gv_okcode.          "OKCODE_ CLEAR , 재사용 방지

      PERFORM get_data.         "DB 를 다시 가져온다.
      PERFORM refresh_grid.     "ALV 화면 갱신

    WHEN 'DISPLAY'.             "레코드를 위에 올려준다.
      CLEAR gv_okcode.

      PERFORM display_data.

    WHEN 'SAVE'.
      CLEAR gv_okcode.

      PERFORM save_data.        " 저장

    WHEN 'CLEAR'.               " gs_data 클리어해준다.
      CLEAR gv_okcode.

      PERFORM clear_data.

    WHEN 'DELETE'.              "삭제
      CLEAR gv_okcode.

      PERFORM delete_data.

  ENDCASE.

ENDMODULE.
