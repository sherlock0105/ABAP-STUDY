FORM set_fcat_layout .

  DATA ls_fcat TYPE lvc_s_fcat.


  gs_layout = VALUE #(
                        zebra      = 'X'
                        cwidth_opt = 'A'
                        sel_mode   = 'D'
                        grid_title = TEXT-t01
                        smalltitle = 'X'
                     ).

  gs_stable = VALUE #(
                       row  = 'X'
                       col  = 'X'
                      ).

  CALL FUNCTION 'LVC_FIELDCATALOG_MERGE'
    EXPORTING
      i_structure_name = 'ZC505T0048'
    CHANGING
      ct_fieldcat      = gt_fcat.


  LOOP AT gt_fcat ASSIGNING FIELD-SYMBOL(<fs_fcat>).

    CASE <fs_fcat>-fieldname.

      WHEN 'AUFNR'.       "핫스팟 (예: 생산오더 클릭)
        <fs_fcat>-hotspot  = 'X'.
        <fs_fcat>-coltext  = 'Production Order'.

      WHEN 'ERDAT' OR 'ERZET' OR 'ERNAM' OR 'AEDAT' OR 'AEZET' OR 'AENAM'.
        <fs_fcat>-no_out = 'X'.

    ENDCASE.

  ENDLOOP.

  gt_fcat = VALUE #( BASE gt_fcat

  ( key       = 'X'
   fieldname = 'ICON'
   coltext   = 'ICON'
   col_pos   = 1
   icon      = 'X')
  ).

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      container_name = 'GCL_CONTAINER'.          " layout 커스텀 컨테이너에 적은 name을 입력

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.                  " Parent Container


  gs_variant-report = sy-repid.

  SET HANDLER : lcl_event_handler=>handle_hotspot_click  FOR gcl_grid,
                lcl_event_handler=>handle_double_click   FOR gcl_grid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant              " Layout
      i_save          = 'A'                     " Save Layout
      i_default       = 'X'                     " Default Display Variant
      is_layout       = gs_layout               " Layout
    CHANGING
      it_outtab       = gt_data                 " Output Table
      it_fieldcatalog = gt_fcat.                " Field Catalog

ENDFORM.

FORM get_data .

  SELECT aufnr, posnr, matnr, pwerk,
         gstrp, gltrp, rsnum, gamng,
         gmein, erdat, erzet, ernam,
         aedat, aezet, aenam,
         CASE
           WHEN aufnr IS NOT INITIAL THEN '@06@'
           ELSE @space
         END AS icon
    FROM zc505t0048
    ORDER BY aufnr, posnr
    INTO CORRESPONDING FIELDS OF TABLE @gt_data.

*  gt_data = CORRESPONDING #( lt_data ).


ENDFORM.

FORM display_data .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid->get_selected_rows     "어디 행을 선택했는지 봐준다.
    IMPORTING
      et_index_rows = gt_row.                 " Indexes of Selected Rows

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH 'First, select row'. "GT_ROW 가 선택하지 않고 DISPLAY 눌렀을때 안내 메세지 및 EXIT
    EXIT.
  ENDIF.

  IF lines( gt_row ) > 1. "다중 행 선택 후 display를 막아준다.
    MESSAGE s000 WITH 'Please select a one row.'.
    EXIT.
  ENDIF.

  READ TABLE gt_row INTO gs_row INDEX 1.     "READ TABLE 로 한행만 가져온다.

  CHECK sy-subrc EQ 0.    "sy-subrc 가 0 아 아니면 아래 로직 수행하지 않는다.

  READ TABLE gt_data INTO gs_data INDEX gs_row-index.

FORM save_data .

  DATA : ls_save TYPE zc505t0048.

  IF gs_data-aufnr IS INITIAL OR
     gs_data-posnr IS INITIAL. "KEY 값이 없으면 EXIT.

    MESSAGE s000 WITH 'Input key field'
    DISPLAY LIKE 'E'.

    EXIT.

  ENDIF.

  IF gs_data-gamng IS NOT INITIAL. "수량이 입력이 되어있는데 단위가 없으면 EXIT 처리 후 안내 메세지.

    IF gs_data-gmein IS INITIAL.

      MESSAGE s000 WITH '단위를 입력하세요' DISPLAY LIKE 'E'.
      EXIT.

    ENDIF.

  ENDIF.

  MOVE-CORRESPONDING gs_data TO ls_save.

  IF ls_save-erdat IS INITIAL.         "신규 데이터인 경우

    ls_save-erdat = sy-datum.          "생성일
    ls_save-erzet = sy-uzeit.          "생성시간
    ls_save-ernam = sy-uname.          "생성자

  ELSE.                                "기존 데이터인 경우

    ls_save-aedat = sy-datum.          "변경일
    ls_save-aezet = sy-uzeit.          "변경시간
    ls_save-aenam = sy-uname.          "변경자

  ENDIF.

  MODIFY zc505t0048 FROM ls_save.  "DB 테이블에 반영 (Insert or Update)

  IF sy-dbcnt > 0.                "변경이 일어나면 commit 아니면 rollback

    COMMIT WORK AND WAIT.
    PERFORM get_data.              "DB 불러오기
    PERFORM refresh_grid.          "화면 재갱신

    MESSAGE s002.                  "Data was saved successfully

  ELSE.

    ROLLBACK WORK.                 "아닐 시 ROLLBACK , 에러메세지 표시
    MESSAGE s004 DISPLAY LIKE 'E'. "Data save failed

  ENDIF.

ENDFORM.

FORM delete_data .

  IF gs_data-aufnr IS INITIAL OR "PK 를 사용하여 DELETE 를 해준다.
     gs_data-posnr IS INITIAL.   "AUFNR 또는 POSNR 이 초기값이면 EXIT.

    MESSAGE s000 WITH 'Required key field'
    DISPLAY LIKE 'E'.

    EXIT.

  ENDIF.

  DELETE FROM zc505t0048
   WHERE aufnr EQ gs_data-aufnr   "KEY 값을 이용해서 DELETE
     AND posnr EQ gs_data-posnr.

  IF sy-subrc EQ 0.     "sy-subrc 가 0이면 commit 아니면 rollback

    COMMIT WORK AND WAIT.
    MESSAGE s003.          "Data was deleted successfully
    PERFORM get_data.      "DB를 받아온다.
    PERFORM refresh_grid.  "화면 갱신

  ELSE.

    ROLLBACK WORK.
    MESSAGE s005 DISPLAY LIKE 'E'. "Data deletion failed

  ENDIF.

ENDFORM.
