FORM get_scarr .

  REFRESH : gt_scarr.

  SELECT carrid carrname currcode url
    INTO CORRESPONDING FIELDS OF TABLE gt_scarr
    FROM scarr
    WHERE carrid IN so_scarr.

ENDFORM.

FORM get_spfli .

  REFRESH gt_spfli.

    INTO CORRESPONDING FIELDS OF TABLE gt_spfli

    WHERE carrid IN so_scarr.

FORM display_tab1 .

  CREATE OBJECT gcl_container_scarr
    EXPORTING
      container_name = 'GCL_CONTAINER_SCARR'. " Name of the Screen CustCtrl Name to Link Container To

  CREATE OBJECT gcl_grid_scarr
    EXPORTING
      i_parent = gcl_container_scarr.          " Parent Container

  gs_variant-report = sy-repid. "이것만 들고 가지고 가면 안된다. subscreen
  gs_variant-handle = '1ST'. "의미있는 이름 x sum 같은거
  "ALV 가 2개 이상인 프로그램 경우 각 그리드의 구분자 역할

  CALL METHOD gcl_grid_scarr->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant                 " Layout
      i_save          = 'A'                 " Save Layout
      i_default       = 'X'              " Default Display Variant
      is_layout       = gs_layout                 " Layout
    CHANGING
      it_outtab       = gt_scarr                 " Output Table
      it_fieldcatalog = gt_fcat_1.               " Field Catalog

ENDFORM.
