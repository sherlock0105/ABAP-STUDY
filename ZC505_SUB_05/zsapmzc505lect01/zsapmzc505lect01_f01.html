FORM set_class_text .

  CLEAR gs_input-class_text.

  CASE gs_input-class.
    WHEN 'CL1'.
      gs_input-class_text = 'Class 1'.
    WHEN 'CL2'.
      gs_input-class_text = 'Class 2'.
    WHEN 'CL3'.
      gs_input-class_text = 'Class 3'.
    WHEN 'CL4'.
      gs_input-class_text = 'Class 4'.
    WHEN 'CL5'.
      gs_input-class_text = 'Class 5'.
    WHEN 'CL6'.
      gs_input-class_text = 'Class 6'.
    WHEN 'CL7'.
      gs_input-class_text = 'Class 7'.
  ENDCASE.

*  MESSAGE i000 WITH gs_input-class_text.

ENDFORM.

FORM set_gender_text .

  CLEAR gs_input-gender_text.

  CASE gs_input-gender.
    WHEN 'M'.
      gs_input-gender_text = 'Male'.
    WHEN 'F'.
      gs_input-gender_text = 'Female'.
  ENDCASE.

  MESSAGE i000 WITH gs_input-gender_text.

ENDFORM.

FORM f4_cl_num .

  DATA : BEGIN OF ls_cl_num,
           class  TYPE zc505t0002-class,
           cl_num TYPE zc505t0002-cl_num,
           name   TYPE zc505t0002-name,
           phone  TYPE zc505t0002-phone,
           gender TYPE zc505t0002-gender,
         END OF ls_cl_num,

         lt_cl_num LIKE TABLE OF ls_cl_num,

         ls_return TYPE ddshretval,
         lt_return LIKE TABLE OF ls_return.

  CLEAR   : ls_cl_num, ls_return.
  REFRESH : lt_cl_num, lt_return.

  SELECT class cl_num name phone gender
    INTO CORRESPONDING FIELDS OF TABLE lt_cl_num
    FROM zc505t0002
   WHERE class EQ gs_input-class.

*      dynprofield  = 'GS_INPUT-CL_NUM'  "더블클릭한 값을 떨굴 화면의 필드
*      window_title = 'Student List'

      return_tab   = lt_return
      value_tab    = lt_cl_num.

  READ TABLE lt_return INTO ls_return INDEX 1. "더블클릭한 행을 읽어온다.

  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

  gs_input-cl_num = ls_return-fieldval.

FORM set_name .

  CLEAR gs_input-name.

  SELECT SINGLE name
    INTO gs_input-name
    FROM zc505t0002
   WHERE class  = gs_input-class
     AND cl_num = gs_input-cl_num.

ENDFORM.
