FORM make_fcat  USING  pv_key
                       pv_field
                       pv_text
                       pv_ref_table
                       pv_ref_field
                       pv_edit.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.
  gs_fcat-edit      = pv_edit.

  CASE pv_field.

    WHEN 'PRODUCT_QTY'. "통화키 연결
      gs_fcat-qfieldname = 'PRODUCT_UNIT'.

    WHEN 'UNIT_PRICE'.
      gs_fcat-cfieldname = 'PRICE_UNIT'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM get_data .

  _clear gs_data gt_data.

FORM save_data .

  DATA : ls_save TYPE zc505t0050.

*key 값 체크
  IF gs_data-steel IS INITIAL OR
     gs_data-werks IS INITIAL.

    MESSAGE s000 WITH 'Input key field data' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  MOVE-CORRESPONDING gs_data TO ls_save. "ls_save 읽은 값 전달

  CALL METHOD gcl_text_edit->get_textstream
    IMPORTING
      text = ls_save-memo.                  " Text as String with Carriage Returns and Linefeeds

  cl_gui_cfw=>flush( ).

  MODIFY zc505t0050 FROM ls_save.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.
    PERFORM get_data.
    PERFORM refresh_grid.

    CALL METHOD gcl_text_edit->delete_text
      EXCEPTIONS
        error_cntl_call_method = 1                " Error while setting property of TextEdit control!
        OTHERS                 = 2.

    MESSAGE s002.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s004 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM delete_data .

  IF gs_data-steel IS INITIAL OR "둘중 키값이 하나라도 없으면 나가라
     gs_data-werks IS INITIAL.
    MESSAGE s000 WITH 'Please input key field' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  DELETE FROM zc505t0050
  WHERE steel EQ gs_data-steel
   AND  werks EQ gs_data-werks.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.
    PERFORM get_data.
    PERFORM refresh_grid.
    MESSAGE s003.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s000 WITH 'Delete failed'.

  ENDIF.

ENDFORM.

FORM display_data .

  _clear gs_row gt_row.

  CALL METHOD gcl_grid_alv->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.                 " Indexes of Selected Rows

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH 'First, select a row'.
    EXIT.
  ENDIF.

  READ TABLE gt_row INTO gs_row INDEX 1.

  CHECK sy-subrc EQ 0.

  READ TABLE gt_data INTO gs_data INDEX gs_row-index.

  CALL METHOD gcl_text_edit->set_textstream
    EXPORTING
      text = gs_data-memo.

ENDFORM.

FORM f4_werks .

  DATA : BEGIN OF ls_data,
           werks TYPE t001w-werks,
           name1 TYPE t001w-name1,
         END OF ls_data,

         lt_data LIKE TABLE OF ls_data.

  SELECT werks name1
    INTO CORRESPONDING FIELDS OF TABLE lt_data
    FROM t001w.

  IF sy-subrc NE 0.
    MESSAGE s001 DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  CALL FUNCTION 'F4IF_INT_TABLE_VALUE_REQUEST'    " F4 help also returning the value to be displayed in internal table
    EXPORTING
      retfield     = 'WERKS'                      " Name of return field in FIELD_TAB
      dynpprog     = sy-repid                     " Current program
      dynpnr       = sy-dynnr                     " Screen number
      dynprofield  = 'GS_DATA-WERKS'              " Name of screen field for value return
      window_title = 'Plants/Branches'            " Title for the hit list
      value_org    = 'S'                          " Value return: C: cell by cell, S: structured
    TABLES
      value_tab    = lt_data.                     " Table of values: entries cell by cell

ENDFORM.
