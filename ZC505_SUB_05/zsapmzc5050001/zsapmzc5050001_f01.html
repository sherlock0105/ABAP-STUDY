FORM get_student .

  CLEAR   gs_input.
  REFRESH gt_input.

  SELECT class cl_num name phone gender
         erdat erzet ernam aedat aezet
         aenam
    INTO CORRESPONDING FIELDS OF TABLE gt_input
    FROM zc505t0002
   ORDER BY class cl_num.

FORM save_data .

  DATA : ls_save TYPE zc505t0002. "한줄만 넣을거라 table 이 필요 없다.
  CLEAR ls_save.

  IF gs_input-class IS INITIAL OR gs_input-cl_num IS INITIAL.
    MESSAGE s000 WITH 'Class and Student no is required field'
    DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  MOVE-CORRESPONDING gs_input TO ls_save.

  IF ls_save-erdat IS INITIAL.

    ls_save-erdat = sy-datum.
    ls_save-erzet = sy-uzeit.
    ls_save-ernam = sy-uname.

  ELSE.

    ls_save-aedat = sy-datum.
    ls_save-aezet = sy-uzeit.
    ls_save-aenam = sy-uname.

  ENDIF.

  MODIFY zc505t0002 FROM ls_save. "테이블이 아님.

  IF sy-dbcnt > 0.
    COMMIT WORK AND WAIT.

    PERFORM get_student.
    PERFORM refresh_grid.

    MESSAGE s002.
  ELSE.
    ROLLBACK WORK.
    MESSAGE s004 DISPLAY LIKE 'E'.
  ENDIF.

ENDFORM.

FORM delete_data .

  IF gs_input-class IS INITIAL OR
     gs_input-cl_num IS INITIAL.

    MESSAGE s000 WITH 'Class and Student no is required field'
    DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  DELETE FROM zc505t0002
   WHERE class = gs_input-class
     AND cl_num = gs_input-cl_num.

  IF sy-subrc EQ 0.
    COMMIT WORK AND WAIT.
    MESSAGE s003.
    PERFORM get_student.
    PERFORM refresh_grid.

  ELSE.
    ROLLBACK WORK.
    MESSAGE s005 DISPLAY LIKE 'E'.
  ENDIF.

ENDFORM.
