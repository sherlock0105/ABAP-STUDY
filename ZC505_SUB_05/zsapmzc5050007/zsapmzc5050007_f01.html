FORM posting .

  DATA : lv_wrbtr_s(20),
         lv_wrbtr_h(20),
         lv_msg(100).

  CLEAR   : gs_bdcdata, gs_messtab, gs_param.
  REFRESH : gt_bdcdata, gt_messtab.

* Option 설정
  gs_param-dismode = 'N'.  " A : 모든 Screen 표시 (개발/디버깅용)
  " N : Screen 미표시 (실운영, 백그라운드)
  " E : 에러 발생 시에만 Screen 표시

  gs_param-nobinpt = 'X'.  " X : Batch Input 세션 생성 안 함 (CALL TRANSACTION)
  " SPACE : Batch Input 세션 생성 (SM35에서 실행)

  gs_param-updmode = 'S'.  " S : Synchronous Update (즉시 DB 반영, 안정적)

  CLEAR : lv_wrbtr_h, lv_wrbtr_s.

"수량은 char 로 변환해서 사용해야함.
"BDC는 화면에 입력되는 문자열 그대로 필요
  WRITE : gs_posting-s_wrbtr TO lv_wrbtr_s CURRENCY gs_posting-waers,
          gs_posting-h_wrbtr TO lv_wrbtr_h CURRENCY gs_posting-waers.
"공백 제거
  CONDENSE lv_wrbtr_s NO-GAPS.
  CONDENSE lv_wrbtr_h NO-GAPS.

  PERFORM make_bdcdata USING : 'X'    'SAPMF05A'    '0100',
                               ' '    'BDC_OKCODE'  '/00',
                               ' '    'BKPF-BLDAT'  sy-datum,
                               ' '    'BKPF-BLART'  gs_posting-blart,
                               ' '    'BKPF-BUKRS'  gs_posting-bukrs,
                               ' '    'BKPF-WAERS'  gs_posting-waers,
                               ' '    'RF05A-NEWBS' gs_posting-s_bschl,
                               ' '    'RF05A-NEWKO' gs_posting-s_hkont,

                               'X'    'SAPMF05A'    '0300',
                               ' '    'BDC_OKCODE'  '/00',
                               ' '    'BSEG-WRBTR'  lv_wrbtr_s,
                               ' '    'RF05A-NEWBS' gs_posting-h_bschl,
                               ' '    'RF05A-NEWKO' gs_posting-h_hkont,

                               'X'    'SAPMF05A'    '0300',
                               ' '    'BDC_OKCODE'  '=BU',
                               ' '    'BSEG-WRBTR'  lv_wrbtr_h.

  CALL TRANSACTION 'FB01' USING         gt_bdcdata
                          OPTIONS FROM  gs_param
                          MESSAGES INTO gt_messtab.

  READ TABLE gt_messtab INTO gs_messtab WITH KEY msgid = 'F5'
                                                 msgnr = '312'.

  IF sy-subrc EQ 0.

    CALL FUNCTION 'MESSAGE_TEXT_BUILD'
      EXPORTING
        msgid               = gs_messtab-msgid
        msgnr               = gs_messtab-msgnr
        msgv1               = gs_messtab-msgv1
        msgv2               = gs_messtab-msgv2
        msgv3               = gs_messtab-msgv3
        msgv4               = gs_messtab-msgv4
      IMPORTING
        message_text_output = lv_msg.

    CONDENSE lv_msg.

    MESSAGE s000 WITH lv_msg.
  ELSE.
    MESSAGE s000 WITH 'Posting failed' DISPLAY LIKE 'E'.
  ENDIF.
ENDFORM.

FORM make_bdcdata  USING  pv_begin pv_fnam pv_fval.

  CASE pv_begin.
    WHEN 'X'.
      gs_bdcdata-dynbegin = pv_begin.
      gs_bdcdata-program  = pv_fnam.
      gs_bdcdata-dynpro   = pv_fval.

    WHEN OTHERS.
      gs_bdcdata-fnam = pv_fnam.
      gs_bdcdata-fval = pv_fval.

  ENDCASE.

  APPEND gs_bdcdata TO gt_bdcdata.
  CLEAR  gs_bdcdata.

ENDFORM.
