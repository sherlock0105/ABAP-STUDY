FORM set_fcat_layout .

  gs_layout = VALUE #(
              zebra      = 'X'
              cwidth_opt = 'A'
              sel_mode   = 'B'
              grid_title = TEXT-t02
              smalltitle = 'X'
                   ).

  gs_stable = VALUE #(
                row = 'X'
                col = 'X'
                     ).

  CALL FUNCTION 'LVC_FIELDCATALOG_MERGE'
    EXPORTING
      i_structure_name   = 'ZC505CDSV0017'                 " Structure name (structure, table, view)
      i_bypassing_buffer = 'X'
    CHANGING
      ct_fieldcat        = gt_fcat.                 " Field Catalog with Field Descriptions

  LOOP AT gt_fcat ASSIGNING FIELD-SYMBOL(<fs_fcat>).

    CASE <fs_fcat>-fieldname.
      WHEN 'GROUP_ID' OR 'MEMBER_ID' OR 'MEMBER_NAME' OR
           'MEMBER_GENDER' OR 'MEMBER_ADDR' OR 'MEMBER_PHONE'.

      WHEN 'ERDAT' OR 'ERZET' OR 'ERNAM' OR 'AEDAT' OR 'AEZET' OR 'AENAM'.
        <fs_fcat>-no_out = 'X'.

    ENDCASE.

  ENDLOOP.

ENDFORM.

FORM save_data .

  DATA : ls_save TYPE zc505t0040.

  IF gs_idol-group_id IS INITIAL OR
     gs_idol-member_Id IS INITIAL.

    MESSAGE s000 WITH 'Input key field' DISPLAY LIKE 'E'.
    EXIT.

  ENDIF.

  IF gs_idol-erdat IS INITIAL.

    gs_idol-erdat = sy-datum.
    gs_idol-erzet = sy-uzeit.
    gs_idol-ernam = sy-uname.

  ELSE.

    gs_idol-aedat = sy-datum.
    gs_idol-aezet = sy-uzeit.
    gs_idol-aenam = sy-uname.

  ENDIF.

  MOVE-CORRESPONDING gs_idol TO ls_save.

  MODIFY zc505t0040 FROM ls_save.

  IF sy-dbcnt > 0.

    COMMIT WORK AND WAIT.
    PERFORM get_data.
    PERFORM refresh_grid.

    MESSAGE s002.

  ELSE.

    ROLLBACK WORK.
    MESSAGE s004 DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.

FORM display_data .

  CLEAR gs_row.
  REFRESH gt_row.

  CALL METHOD gcl_grid->get_selected_rows
    IMPORTING
      et_index_rows = gt_row.                 " Indexes of Selected Rows

  IF gt_row IS INITIAL.
    MESSAGE s000 WITH 'First, select row'.
    EXIT.
  ENDIF.

  IF lines( gt_row ) > 1.
    MESSAGE s000 WITH 'Please select a one row.'.
    EXIT.
  ENDIF.

  READ TABLE gt_row INTO gs_row INDEX 1.

  CHECK sy-subrc EQ 0.

  READ TABLE gt_idol INTO gs_idol INDEX gs_row-index.

ENDFORM.
