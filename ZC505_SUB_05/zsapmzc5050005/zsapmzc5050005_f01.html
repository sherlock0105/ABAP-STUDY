    EXPORTING

   WHERE seq = gv_seq.

  IF sy-subrc NE 0.

* 문자열 나누기 : SPLIT ..
* 1. SPLIT <변수> AT <구분자> INTO <변수1> <변수2> ...<변수N>.
* --> 구분자에 의해서 나눠지는 문자열들을 차례대로 <변수1> ~ <변수N> 까지 넣어준다.
*     INTO 이후에 나열된 변수보다 많은 경우 나머지 문자열은 무시된다.
*     나눠지는 문자열이 나열된 변수보다 적은 경우 나머지 변수 무시된다.
*2. SPLIT <변수> AT <구분자> INTO TABLE <ITAB>
* --> 구분자에 의해서 나눠지는 문자열들을 <ITAB> 에 차례대로 APPEND 해준다.

FORM save_text_stream .

  DATA : ls_save TYPE zc505t0049,

         BEGIN OF ls_text,
           line TYPE tdline,
         END OF ls_text,

         lt_text LIKE TABLE OF ls_text.

  CLEAR   ls_text.
  REFRESH lt_text.


  IF gv_seq IS INITIAL.
    MESSAGE s000 WITH 'Input Sequence' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  CALL METHOD gcl_text_l->get_text_as_stream
    IMPORTING
      text = lt_text.                 " text as stream with carrige retruns and linefeeds

  IF lt_text IS INITIAL.
    MESSAGE s000 WITH 'Input description' DISPLAY LIKE 'E'.
    EXIT.
  ENDIF.

  ls_save-seq = gv_seq.

  LOOP AT lt_text INTO ls_text.

    ls_save-description = ls_save-description && ls_text-line.

  ENDLOOP.

  MODIFY zc505t0049 FROM ls_save.

  IF sy-dbcnt > 0.
    COMMIT WORK AND WAIT.
    MESSAGE s002.
  ENDIF.

ENDFORM.
