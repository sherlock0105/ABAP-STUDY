FORM get_airline .

  _clear : gs_scarr gt_scarr,
           gs_spfli gt_spfli.

  SELECT carrid carrname currcode url
    INTO CORRESPONDING FIELDS OF TABLE gt_scarr
    FROM scarr.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM display_split .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left     " Side to Which Control is Docked
      extension = 5000.               " Control Extension

  CREATE OBJECT gcl_splitter
    EXPORTING
      parent  = gcl_container                   " Parent Container
      rows    = 1                   " Number of Rows to be displayed
      columns = 2.                   " Number of Columns to be Displayed

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1                  " Row
      column    = 1                 " Column
    RECEIVING
      container = gcl_container_up.                 " Container

  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1                  " Row
      column    = 2                 " Column
    RECEIVING
      container = gcl_container_down.                 " Container

  CREATE OBJECT gcl_grid_up
    EXPORTING
      i_parent = gcl_container_up.                  " Parent Container

  CREATE OBJECT gcl_grid_down
    EXPORTING
      i_parent = gcl_container_down.                  " Parent Container

  CALL METHOD gcl_splitter->set_column_width
    EXPORTING
      id    = 1                   " Row ID
      width = 60.                 " Height

  SET HANDLER : lcl_event_handler=>handle_double_click FOR gcl_grid_up.

  gs_variant-report = sy-repid.
  gs_variant-handle = 'UP'.

  CALL METHOD gcl_grid_up->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant                  " Layout
      i_save          = 'A'                 " Save Layout
      i_default       = 'X'              " Default Display Variant
      is_layout       = gs_layout                " Layout
    CHANGING
      it_outtab       = gt_scarr                 " Output Table
      it_fieldcatalog = gt_fcat_up.                 " Field Catalog

  gs_variant-handle = 'DN'.

  CALL METHOD gcl_grid_down->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant                  " Layout
      i_save          = 'A'                 " Save Layout
      i_default       = 'X'              " Default Display Variant
      is_layout       = gs_layout                " Layout
    CHANGING
      it_outtab       = gt_spfli                 " Output Table
      it_fieldcatalog = gt_fcat_down.                 " Field Catalog

ENDFORM.
*&---------------------------------------------------------------------*
*& Form handle_double_click
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW
*&      --> E_COLUMN
*&---------------------------------------------------------------------*
FORM handle_double_click  USING    ps_row TYPE lvc_s_row
                                   ps_col TYPE lvc_s_col.

  IF ps_row-rowtype IS NOT INITIAL.
    EXIT.
  ENDIF.

  READ TABLE gt_scarr INTO gs_scarr INDEX ps_row-index.

  IF sy-subrc NE 0.
    EXIT.
  ENDIF.

  PERFORM get_schedule USING gs_scarr-carrid.

  IF gt_spfli IS INITIAL.
    MESSAGE s000 WITH gs_scarr-carrid 'Schedule not found'.
    EXIT.
  ENDIF.

  PERFORM refresh_grid.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form get_schedule
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> GS_SCARR_CARRID
*&---------------------------------------------------------------------*
FORM get_schedule  USING  pv_carrid TYPE scarr-carrid.

  _clear gs_spfli gt_spfli.

  SELECT carrid connid airpfrom airpto deptime arrtime
    INTO CORRESPONDING FIELDS OF TABLE gt_spfli
    FROM spfli
   WHERE carrid EQ pv_carrid.

ENDFORM.
