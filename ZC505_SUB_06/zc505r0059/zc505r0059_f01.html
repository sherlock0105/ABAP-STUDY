FORM display_split .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left   " Side to Which Control is Docked
      extension = 5000.                         " Control Extension
*상위 컨테이너를 나눈다.
  CREATE OBJECT gcl_splitter
    EXPORTING
      parent  = gcl_container                   " Parent Container
      rows    = 3                   " 몇 개의 가로(행) 분할할 것인지
      columns = 1.                  " 몇 개의 세로(열) 분할할 것인지

*1행 1열  컨테이너  .
  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 1                 " Row
      column    = 1                " Column
    RECEIVING
      Container = gcl_container_top. "gt_scarr

*2행 1열  컨테이너  .
  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 2                 " Row
      column    = 1                " Column
    RECEIVING
      Container = gcl_container_bot. "gt_scarr

*3행 1열  컨테이너  .
  CALL METHOD gcl_splitter->get_container
    EXPORTING
      row       = 3                 " Row
      column    = 1                " Column
    RECEIVING
      Container = gcl_container_mid. "gt_sflight

*1행 1열  grid   .
  CREATE OBJECT gcl_grid_top
    EXPORTING
      i_parent = gcl_container_top.                 " Parent Container
*1행 1열  grid  .
  CREATE OBJECT gcl_grid_bot
    EXPORTING
      i_parent = gcl_container_bot.                 " Parent Container

*1행 1열  grid  .
  CREATE OBJECT gcl_grid_mid
    EXPORTING
      i_parent = gcl_container_mid.                 " Parent Container

  CALL METHOD gcl_splitter->set_row_height
    EXPORTING
      id     = 1                   " Row ID
      height = 30.                 " Height

  CALL METHOD gcl_splitter->set_row_height
    EXPORTING
      id     = 2                   " Row ID
      height = 20.                 " Height

  gs_variant-report = sy-repid.
  gs_variant-handle = 'TOP'.       "구분자

  CALL METHOD gcl_grid_top->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant            " Layout
      i_save          = 'A'                   " Save Layout
      i_default       = 'X'                   " Default Display Variant
      is_layout       = gs_layout             " Layout
    CHANGING
      it_outtab       = gt_scarr              " Output Table
      it_fieldcatalog = gt_fcat_top.          " Field Catalog

  gs_variant-handle = 'BOT'.       "구분자

  CALL METHOD gcl_grid_bot->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant            " Layout
      i_save          = 'A'                   " Save Layout
      i_default       = 'X'                   " Default Display Variant
      is_layout       = gs_layout             " Layout
    CHANGING
      it_outtab       = gt_spfli              " Output Table
      it_fieldcatalog = gt_fcat_bot.          " Field Catalog

  gs_variant-handle = 'MID'.       "구분자

  CALL METHOD gcl_grid_mid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant            " Layout
      i_save          = 'A'                   " Save Layout
      i_default       = 'X'                   " Default Display Variant
      is_layout       = gs_layout             " Layout
    CHANGING
      it_outtab       = gt_sflight              " Output Table
      it_fieldcatalog = gt_fcat_mid.          " Field Catalog

ENDFORM.

FORM get_sflight .

  REFRESH : gt_sflight.

  SELECT carrid connid fldate price currency
    INTO CORRESPONDING FIELDS OF TABLE gt_sflight
    FROM sflight
   WHERE carrid IN so_carr
   ORDER BY carrid connid fldate.

ENDFORM.
