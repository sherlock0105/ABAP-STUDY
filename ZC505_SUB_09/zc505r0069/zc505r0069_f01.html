FORM get_product .
  REFRESH : gt_product.

  DATA : lcl_connection TYPE REF TO cl_sql_connection,
         lcl_statement  TYPE REF TO cl_sql_statement,
         lcl_result     TYPE REF TO cl_sql_result_set,
         lr_data        TYPE REF TO data,
         lr_param       TYPE REF TO data,
         lv_sql         TYPE string,
         lx_sql_except  TYPE REF TO cx_root,
         lv_msg         TYPE string.

  TRY.

      lcl_connection = cl_sql_connection=>get_connection( ).

      CREATE OBJECT lcl_statement
        EXPORTING
          con_ref = lcl_connection.

      lv_sql = | SELECT a.AUFNR, a.AUART, a.BUKRS, a.WERKS, c.MATNR,         |
            && |        b.GSTRP, c.POSNR, b.GLTRP, b.RSNUM, c.PSMNG, c.AMEIN |
            && |   FROM AUFK AS a                                            |
            && |  INNER JOIN afko as b                                       |
            && |  on a.aufnr = b.aufnr                                       |
            && |  INNER JOIN afpo as c                                       |
            && |  on a.aufnr = c.aufnr                                       |
            && |  WHERE a.MANDT = ?                                          | "? 은 파라미터 자리가 된다.
            && |    AND a.BUKRS = ?                                          | .

      "동적 할당이 가능하다. placeholder,
      GET REFERENCE OF sy-mandt INTO lr_param. "lr_parma 이 mandt 포인터, 그래서 set+
      lcl_statement->set_param( lr_param ). "첫번째 파마리터 자리에 넣어준다.

      GET REFERENCE OF pa_buk INTO lr_param.
      lcl_statement->set_param( lr_param ).
       "받아올 장소를 저장한다.
      lcl_result = lcl_statement->execute_query( lv_sql ).
      "lr_data, 메모리 주소 공유, 저장된 데이터만 뽑아서 온다. get_refrence /  set_parameter 가 끌어와준다.
      GET REFERENCE OF gt_product INTO lr_data.

      lcl_result->set_param_table( lr_data ).
      lcl_result->next_package( ).

      lcl_result->close( ).

    CATCH cx_sql_exception INTO lx_sql_except.

      lv_msg = lx_sql_except->get_text( ).

      IF lcl_connection IS BOUND.
        TRY.
            lcl_connection->close( ).
          CATCH cx_sql_exception.

        ENDTRY.
      ENDIF.

      MESSAGE s000 WITH lv_msg DISPLAY LIKE 'E'.
      EXIT.

  ENDTRY.

ENDFORM.

*&---------------------------------------------------------------------*
*& Form make_fcat_product
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat_product  USING pv_key
                              pv_field
                              pv_text
                              pv_ref_table
                              pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE gs_fcat-fieldname.
    WHEN 'PSMNG'.
      gs_fcat-qfieldname = 'AMEINS'.
  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.
