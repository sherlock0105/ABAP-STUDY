MODULE display_sbook_screen OUTPUT.

  IF gcl_container_sbook IS NOT BOUND.

    CREATE OBJECT gcl_container_sbook
      EXPORTING
        container_name = 'GCL_CONTAINER_SBOOK'.

    CREATE OBJECT gcl_grid_sbook
      EXPORTING
        i_parent = gcl_container_sbook.

    CALL METHOD gcl_grid_sbook->set_table_for_first_display
      EXPORTING
        is_layout       = gs_layout
      CHANGING
        it_outtab       = gt_sbook
        it_fieldcatalog = gt_fcat_sbook.

  ENDIF.

ENDMODULE.
