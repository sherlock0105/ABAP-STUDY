FORM get_data .

  RANGES : lr_carrid FOR scarr-carrid,
           lr_connid FOR sflight-connid,
           lr_fldate FOR sflight-fldate.

  REFRESH : lr_carrid, lr_connid, lr_fldate, gt_data.

  IF pa_car IS NOT INITIAL.

    lr_carrid[] = VALUE #(
                  (
                  sign   = 'I'
                  option = 'EQ'
                  low    = pa_car
                  )
                        ).

  ENDIF.

  IF pa_con IS NOT INITIAL.

    lr_connid[] = VALUE #(
                    (
                  sign   = 'I'
                  option = 'EQ'
                  low    = pa_con
                    )
                        ).

  ENDIF.

  IF pa_fld IS NOT INITIAL.

    lr_fldate-sign   = 'I'.
    lr_fldate-option = 'EQ'.
    lr_fldate-low    = pa_fld.

    APPEND lr_fldate.
  ENDIF.

  SELECT status, carrid, connid, fldate, carrname,
         cityfrom, cityto, price, currcode
    INTO CORRESPONDING FIELDS OF TABLE @gt_data
    FROM zc505cds0014
*      UP TO 300 ROWS
   WHERE carrid IN @lr_carrid
     AND connid IN @lr_connid
     AND fldate IN @lr_fldate
   ORDER BY carrid, connid, fldate.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left " Side to Which Control is Docked
      extension = 5000.                       " Control Extension

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  SET HANDLER : lcl_event_handler=>handle_hotspot       FOR gcl_grid,
                lcl_event_handler=>handle_double_click  FOR gcl_grid.

  CALL METHOD gcl_grid->register_edit_event
    EXPORTING
      i_event_id = gcl_grid->mc_evt_modified.                 " Event ID

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant       " Layout
      i_save          = 'A'              " Save Layout
      i_default       = 'X'              " Default Display Variant
      is_layout       = gs_layout        " Layout
    CHANGING
      it_outtab       = gt_data           " Output Table
      it_fieldcatalog = gt_fcat.         " Field Catalog

ENDFORM.

FORM handle_hotspot  USING    ps_row TYPE lvc_s_row
                              ps_col TYPE lvc_s_col.

  CHECK ps_row-rowtype IS INITIAL.

  READ TABLE gt_data INTO DATA(gs_data) INDEX ps_row-index.

  CHECK sy-subrc EQ 0.

  PERFORM get_sbook_data USING gs_data-carrid
                               gs_data-connid
                               gs_data-fldate.

  IF gt_sbook IS INITIAL.
    MESSAGE s000 WITH 'Data not found'.
    EXIT.
  ENDIF.

  CALL SCREEN '0101' STARTING AT 20 3.




ENDFORM.

FORM handle_double_click  USING     ps_row TYPE lvc_s_row
                                    ps_col TYPE lvc_s_col.

  CHECK ps_row-rowtype IS INITIAL.

  READ TABLE gt_data INTO DATA(gs_data) INDEX ps_row-index.

  CHECK sy-subrc EQ 0.

  CASE ps_col-fieldname.
    WHEN 'FLDATE'.
      PERFORM get_sflight USING gs_data-carrid
                                gs_data-connid
                                gs_data-fldate.

      CALL SCREEN '0102' STARTING AT 20 3.

    WHEN OTHERS.
      EXIT.

  ENDCASE.

  IF gt_sflight IS INITIAL.
    MESSAGE s000 WITH 'Data not found'.
    EXIT.
  ENDIF.

ENDFORM.

FORM set_fcat_layout_pop .

  gs_layout = VALUE #( BASE gs_layout
            grid_title = 'SBOOK INFO'
            smalltitle = 'X'
            no_toolbar = 'X'
                 ).

  CALL FUNCTION 'LVC_FIELDCATALOG_MERGE'
    EXPORTING
      i_structure_name = 'SBOOK'
    CHANGING
      ct_fieldcat      = gt_fcat_sbook.

  LOOP AT gt_fcat_sbook ASSIGNING FIELD-SYMBOL(<fs_fcat>).

    CASE <fs_fcat>-fieldname.

      WHEN 'CARRID' OR 'CONNID' OR 'FLDATE' OR 'BOOKID'.
        <fs_fcat>-key = 'X'.

      WHEN 'LUGGWEIGHT' OR 'WUNIT' OR 'FORCURAM' OR 'FORCURKEY'.

      WHEN 'INVOICE'.
        <fs_fcat>-checkbox = 'X'.

      WHEN OTHERS.
        <fs_fcat>-no_out = 'X'.

    ENDCASE.

  ENDLOOP.

ENDFORM.

* Structure 형태로 받으려고 set_param_struct 메소드 호출함
