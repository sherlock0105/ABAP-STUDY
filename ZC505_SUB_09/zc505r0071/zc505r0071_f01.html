*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING     pv_key
                          pv_field
                          pv_text
                          pv_ref_table
                          pv_ref_field.


  gs_fcat = VALUE #(
            key       = pv_key
            fieldname = pv_field
            coltext   = pv_text
            ref_table = pv_ref_table
            ref_field = pv_ref_field
                   ).

  CASE pv_field.

    WHEN 'WERKS'.
      gs_fcat-hotspot = 'X'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.
  CLEAR  gs_fcat.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left " Side to Which Control is Docked
      extension = 5000.                       " Control Extension

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  SET HANDLER : lcl_event_handler=>handle_hotspot       FOR gcl_grid,
                lcl_event_handler=>handle_double_click  FOR gcl_grid.

  CALL METHOD gcl_grid->register_edit_event
    EXPORTING
      i_event_id = gcl_grid->mc_evt_modified.                 " Event ID

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant       " Layout
      i_save          = 'A'              " Save Layout
      i_default       = 'X'              " Default Display Variant
      is_layout       = gs_layout        " Layout
    CHANGING
      it_outtab       = gt_data           " Output Table
      it_fieldcatalog = gt_fcat.         " Field Catalog

ENDFORM.

*& Form handle_hotspot
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW_ID
*&      --> E_COLUMN_ID
*&---------------------------------------------------------------------*
FORM handle_hotspot  USING  ps_row TYPE lvc_s_row
                            ps_col TYPE lvc_s_col.

  REFRESH : gt_marc.

  CHECK ps_row-rowtype IS INITIAL.

*  READ TABLE gt_data INTO DATA(gs_data) INDEX ps_row-index.

  DATA(ls_data) = VALUE #( gt_data[ ps_row-index ] OPTIONAL ).

  CHECK ls_data IS NOT INITIAL.

  CHECK ps_col-fieldname EQ 'WERKS'.

  PERFORM get_marc_data USING ls_data-matnr
                              ls_data-werks.

  IF gt_marc IS INITIAL.
    MESSAGE s000 WITH 'Data not found'.
    EXIT.
  ENDIF.

  CALL SCREEN '0101' STARTING AT 20 3.

ENDFORM.

*& Form get_mara_data
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> LS_DATA_MATNR
*&---------------------------------------------------------------------*
FORM get_mara_data  USING  pv_matnr TYPE mara-matnr.

  REFRESH: gt_mara.

  DATA : lcl_connection TYPE REF TO cl_sql_connection, "DB 연결
         lcl_statement  TYPE REF TO cl_sql_statement, "SQL 실행
         lcl_result     TYPE REF TO cl_sql_result_set, "결과 받기
         lr_data        TYPE REF TO data,
         lv_sql         TYPE string,
         lx_sql_except  TYPE REF TO cx_root,
         lv_msg         TYPE string.

  TRY.
      lcl_connection = cl_sql_connection=>get_connection( ). " DB CONNECTION 정보 얻기

      CREATE OBJECT lcl_statement
        EXPORTING
          con_ref = lcl_connection.                          " Database Connection

      lv_sql =   | SELECT A.MATNR, B.MAKTX, A.MTART, C.MTBEZ, A.MATKL, D.MBBEZ | &&
                 |   FROM MARA AS A                                            | &&
                 |   LEFT OUTER JOIN MAKT AS B                                 | &&
                 |     ON A.MATNR = B.MATNR                                    | &&
                 |    AND B.SPRAS = '{ sy-langu }'                             | &&
                 |    AND B.MANDT = '{ sy-mandt }'                             | &&
                 |   LEFT OUTER JOIN T134T AS C                                | &&
                 |     ON A.MTART = C.MTART                                    | &&
                 |    AND C.SPRAS = '{ sy-langu }'                             | &&
                 |    AND C.MANDT = '{ sy-mandt }'                             | &&
                 |   LEFT OUTER JOIN T137T AS D                                | &&
                 |     ON A.MBRSH = D.MBRSH                                    | &&
                 |    AND D.SPRAS = '{ sy-langu }'                             | &&
                 |    AND D.MANDT = '{ sy-mandt }'                             | &&
                 |  WHERE A.MANDT  = '{ sy-mandt  }'                           | &&
                 |    AND A.MATNR  = '{ pv_matnr }'                            |.

      lcl_result = lcl_statement->execute_query( lv_sql ).

      GET REFERENCE OF gt_mara INTO lr_data.

      lcl_result->set_param_table( lr_data ).


      lcl_result->next_package( ).

      lcl_result->close( ).

    CATCH cx_sql_exception INTO lx_sql_except.

      lv_msg = lx_sql_except->get_text( ).
      lcl_result->close( ).

      MESSAGE s000 WITH lv_msg DISPLAY LIKE 'E'.
      EXIT.

  ENDTRY.

ENDFORM.
