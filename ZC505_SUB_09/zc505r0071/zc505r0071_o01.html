MODULE display_screen_pop OUTPUT.

  IF gcl_container_marc IS NOT BOUND.

    CREATE OBJECT gcl_container_marc
      EXPORTING
        container_name = 'GCL_CONTAINER_MARC'.

    CREATE OBJECT gcl_grid_marc
      EXPORTING
        i_parent = gcl_container_marc.

    CALL METHOD gcl_grid_marc->set_table_for_first_display
      EXPORTING
        is_layout       = gs_layout
      CHANGING

  ENDIF.
