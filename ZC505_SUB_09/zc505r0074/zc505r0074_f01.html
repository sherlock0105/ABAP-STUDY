  gs_stable = VALUE #(
                row = 'X'

                     ).

  CALL FUNCTION 'LVC_FIELDCATALOG_MERGE'
    EXPORTING
      i_structure_name = 'ZC505T0022'                 " Structure name (structure, table, view)
    CHANGING
      ct_fieldcat      = gt_fcat.

  LOOP AT gt_fcat ASSIGNING FIELD-SYMBOL(<fs_fcat>).

    CASE <fs_fcat>-fieldname.

      WHEN 'AUART'.

        <fs_fcat>-edit = ' '.

            OR 'AEDAT' OR 'AEZET' OR 'AENAM'.

        <fs_fcat>-no_out = 'X'.

      WHEN OTHERS.

        <fs_fcat>-edit = 'X'.

    ENDCASE.

  ENDLOOP.

FORM save_data .

  DATA : lv_commit,
         lt_save   TYPE TABLE OF zc505t0022,
         ls_save   LIKE LINE OF lt_save,
         lt_del    TYPE TABLE OF zc505t0022.

  CALL METHOD gcl_grid->check_changed_data.

  IF gt_mat_del IS NOT INITIAL.

    lt_del = CORRESPONDING #( gt_mat_del ).

    DELETE zc505t0022 FROM TABLE lt_del.

    CHECK sy-subrc EQ 0.
    lv_commit = 'X'.

  ENDIF.

  LOOP AT gt_mat ASSIGNING FIELD-SYMBOL(<fs_mat>) WHERE mark IS NOT INITIAL.

    IF <fs_mat>-matnr IS INITIAL OR <fs_mat>-auart IS INITIAL.
      MESSAGE s000 WITH 'Input key field' DISPLAY LIKE 'E'.
      CONTINUE.
    ENDIF.

    ls_save = CORRESPONDING zc505t0022( <fs_mat> ).

    IF ls_save-erdat IS INITIAL. " 신규 데이터
      ls_save-erdat = sy-datum.
      ls_save-erzet = sy-uzeit.
      ls_save-ernam = sy-uname.
    ELSE.                        " 기존 데이터 수정
      ls_save-aedat = sy-datum.
      ls_save-aezet = sy-uzeit.
      ls_save-aenam = sy-uname.
    ENDIF.

    APPEND ls_save TO lt_save.
    CLEAR  ls_save.

  ENDLOOP.

  IF lt_save IS NOT INITIAL.
    MODIFY zc505t0022 FROM TABLE lt_save.

    IF sy-dbcnt > 0.
      lv_commit = 'X'.
    ENDIF.

  ENDIF.

  IF lv_commit IS NOT INITIAL.

    COMMIT WORK AND WAIT.

    REFRESH gt_mat_del.

    PERFORM clear_mark.
    MESSAGE s002.

  ELSE.
    ROLLBACK WORK.
    MESSAGE s000 WITH 'Data was not changed' DISPLAY LIKE 'E'.

  ENDIF.

ENDFORM.
