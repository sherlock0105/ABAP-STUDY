FORM get_data .

  CALL METHOD zc505cl_amdp_02=>get_bkpf
    EXPORTING
      iv_bukrs = pa_bukrs
      iv_gjahr = pa_gjahr
    IMPORTING
      et_bkpf  = DATA(lt_data).


  IF lt_data IS INITIAL.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.

  ELSE.

    MOVE-CORRESPONDING lt_data TO gt_data.

  ENDIF.

ENDFORM.

FORM set_color .

  LOOP AT gt_data ASSIGNING FIELD-SYMBOL(<lfs_data>).

    IF <lfs_data>-bschl = 40.
      <lfs_data>-color = VALUE lvc_t_scol(
                            ( fname = 'BSCHL'
                              color = VALUE #( col = 5 int = 1 inv = 0 ) )
                                ).

    ENDIF.

    IF <lfs_data>-pswsl EQ 'EUR'.
      <lfs_data>-pswsl = 'KRW'.
    ENDIF.

  ENDLOOP.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING   pv_key
                        pv_field
                        pv_text
                        pv_ref_table
                        pv_ref_field.


  DATA(gs_fcat) = VALUE lvc_s_fcat(
                      key       = pv_key
                      fieldname = pv_field
                      coltext   = pv_text
                      ref_table = pv_ref_table
                      ref_field = pv_ref_field
                             ).

  CASE pv_field.

    WHEN 'DMBTR'.
      gs_fcat-cfieldname = 'PSWSL'.

    WHEN 'BELNR'.
      gs_fcat-hotspot = 'X'.

  ENDCASE.

  gt_fcat = VALUE #( BASE gt_fcat ( gs_fcat ) ).

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left     " Side to Which Control is Docked
      extension = 5000.               " Control Extension

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.                  " Parent Container

  gs_variant-report = sy-repid.

  SET HANDLER lcl_event_handler=>handle_hotspot FOR gcl_grid.

  CALL METHOD gcl_grid->register_edit_event
    EXPORTING
      i_event_id = gcl_grid->mc_evt_modified.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant       " Layout
      i_save          = 'A'              " Save Layout
      i_default       = 'X'              " Default Display Variant
      is_layout       = gs_layout        " Layout
    CHANGING
      it_outtab       = gt_data           " Output Table
      it_fieldcatalog = gt_fcat.         " Field Catalog

ENDFORM.
*&---------------------------------------------------------------------*
*& Form handle_hotspot
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> E_ROW_ID
*&      --> E_COLUMN_ID
*&---------------------------------------------------------------------*
FORM handle_hotspot  USING    ps_row TYPE lvc_s_row
                              ps_col TYPE lvc_s_col.

  CHECK ps_row-rowtype IS INITIAL.

  READ TABLE gt_data INTO DATA(gs_data) INDEX ps_row-index.

  CHECK sy-subrc EQ 0.

  CASE ps_col-fieldname.

    WHEN 'BELNR'.

      SET PARAMETER ID :
        'BLN' FIELD gs_data-belnr,
        'BUK' FIELD pa_bukrs,
        'GJR' FIELD pa_gjahr.

      CALL TRANSACTION 'FB03' AND SKIP FIRST SCREEN.

  ENDCASE.

ENDFORM.
