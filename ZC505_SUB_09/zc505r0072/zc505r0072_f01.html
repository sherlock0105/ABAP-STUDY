FORM get_sflight .

  REFRESH: gt_sflight.

  DATA : lcl_statement TYPE REF TO cl_sql_statement.

  TRY.
      DATA(lcl_connection) = cl_sql_connection=>get_connection( ). " DB CONNECTION 정보 얻기

      CREATE OBJECT lcl_statement
        EXPORTING
          con_ref = lcl_connection.                          " Database Connection

      DATA(lv_sql) =   | SELECT *                          | &&
                       |   FROM SFLIGHT                    | &&
                       |  WHERE MANDT  = '{ sy-mandt  }'   | &&
                       |    AND CARRID  = '{ pa_car }'     | &&
                       |    AND CONNID  = '{ pa_con }'     | &&
                       |    AND FLDATE  = '{ pa_fld }'     |.


      DATA(lcl_result) = lcl_statement->execute_query( lv_sql ).

      GET REFERENCE OF gt_sflight INTO DATA(lr_data).

      lcl_result->set_param_table( lr_data ).

      lcl_result->next_package( ).

      lcl_result->close( ).

    CATCH cx_sql_exception INTO DATA(lx_sql_except).

      DATA(lv_msg) = lx_sql_except->get_text( ).
      lcl_result->close( ).

      MESSAGE s000 WITH lv_msg DISPLAY LIKE 'E'.
      EXIT.

  ENDTRY.

ENDFORM.

*& Form make_fcat_sbook
*&---------------------------------------------------------------------*

*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_

FORM make_fcat_sbook  USING   pv_key
                              pv_field
                              pv_text
                              pv_ref_table
                              pv_ref_field.

  gs_fcat-key       = pv_key.
  gs_fcat-fieldname = pv_field.
  gs_fcat-coltext   = pv_text.
  gs_fcat-ref_table = pv_ref_table.
  gs_fcat-ref_field = pv_ref_field.

  CASE pv_field.

    WHEN 'LUGGWEIGHT'.
      gs_fcat-qfieldname = 'WUNIT'.

    WHEN 'FORCURAM'.
      gs_fcat-cfieldname = 'FORCURKEY'.

    WHEN 'INVOICE'.
      gs_fcat-checkbox = 'X'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat_2.
  CLEAR  gs_fcat.


ENDFORM.

FORM set_desc .

  DATA : lv_value TYPE dd07v-domvalue_l,
         lv_text  TYPE dd07v-ddtext.


  LOOP AT gt_sbook ASSIGNING FIELD-SYMBOL(<fs_sbook>).

    lv_value = <fs_sbook>-class.

    CALL FUNCTION 'C_DIC_DOMAIN_VALUE_TEXT_READ'
      EXPORTING
        name  = 'S_CLASS'
        spras = sy-langu
        value = lv_value
      IMPORTING
        text  = lv_text.

    <fs_sbook>-class_desc = lv_text.

  ENDLOOP.

ENDFORM.

FORM set_desc_test .

  DATA lt_dom TYPE TABLE OF dd07v.

  CALL FUNCTION 'DD_DOMVALUES_GET'
    EXPORTING
      domname   = 'S_CLASS'
      text      = 'X'
      langu     = sy-langu
    TABLES
      dd07v_tab = lt_dom.

*DOMVALUE_L	도메인 값
*DDTEXT	설명 텍스트

  LOOP AT gt_sbook ASSIGNING FIELD-SYMBOL(<fs_sbook>).

    <fs_sbook>-class_desc =
      VALUE #( lt_dom[ domvalue_l = <fs_sbook>-class ]-ddtext OPTIONAL ).

  ENDLOOP.

ENDFORM.
