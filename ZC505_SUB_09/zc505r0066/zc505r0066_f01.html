*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat_b  USING  pv_key
                         pv_field
                         pv_text
                         pv_ref_table
                         pv_ref_field.

  gs_fcat = VALUE #(
                    key       = pv_key
                    fieldname = pv_field
                    coltext   = pv_text
                    ref_table = pv_ref_table
                    ref_field = pv_ref_field
                    ).


  CASE pv_field.
    WHEN 'LUGGWEIGHT'.
      gs_fcat-qfieldname = 'WUNIT'.
    WHEN 'FORCURAM'.
      gs_fcat-cfieldname = 'FORCURKEY'.
  ENDCASE.


  APPEND gs_fcat TO gt_fcat_b.
  CLEAR  gs_fcat.

ENDFORM.
