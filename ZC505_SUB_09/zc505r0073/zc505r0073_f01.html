FORM get_data .

  SELECT carrid connid fldate
         paymentsum currency planetype
    INTO CORRESPONDING FIELDS OF TABLE gt_data
    FROM sflight
   WHERE carrid IN so_car
     AND connid IN so_con.

  IF gt_data IS INITIAL.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM set_fcat_layout .

  gs_layout = VALUE #(
                      zebra      = 'X'
                      cwidth_opt = 'A'
                      sel_mode   = 'D'
                      ).


  gs_stable = VALUE #(
                      col = 'X'
                      row = 'X'
                      ).

  PERFORM make_fcat USING :
      'X' 'CARRID    ' ' ' 'SFLIGHT'  ' ',
      'X' 'CONNID    ' ' ' 'SFLIGHT'  ' ',
      'X' 'FLDATE    ' ' ' 'SFLIGHT'  ' ',
      ' ' 'PAYMENTSUM' ' ' 'SFLIGHT'  ' ',
      ' ' 'CURRENCY  ' ' ' 'SFLIGHT'  ' ',
      ' ' 'PLANETYPE ' ' ' 'SFLIGHT'  ' '.

ENDFORM.
*&---------------------------------------------------------------------*
*& Form make_fcat
*&---------------------------------------------------------------------*
*& text
*&---------------------------------------------------------------------*
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&      --> P_
*&---------------------------------------------------------------------*
FORM make_fcat  USING  pv_key
                       pv_field
                       pv_text
                       pv_ref_table
                       pv_ref_field.

  DATA(gs_fcat) = VALUE lvc_s_fcat(
                         key       = pv_key
                         fieldname = pv_field
                         coltext   = pv_text
                         ref_table = pv_ref_table
                         ref_field = pv_ref_field
                                  ).

  CASE pv_field.

    WHEN 'PAYMENTSUM'.
      gs_fcat-cfieldname = 'CURRENCY'.

  ENDCASE.

  APPEND gs_fcat TO gt_fcat.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left     " Side to Which Control is Docked
      extension = 5000.                           " Control Extension

  CREATE OBJECT gcl_gird
    EXPORTING
      i_parent = gcl_container.                  " Parent Container

  CALL METHOD gcl_gird->set_table_for_first_display
    EXPORTING
      i_save          = 'A'                     " Save Layout
      is_layout       = gs_layout               " Layout
    CHANGING
      it_outtab       = gt_data                 " Output Table
      it_fieldcatalog = gt_fcat.                " Field Catalog

ENDFORM.
