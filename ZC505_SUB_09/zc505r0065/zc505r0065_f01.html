FORM get_data .

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE @gt_data
    FROM zc505cds0005
    UP TO 50 ROWS
   WHERE carrier     EQ @pa_car
     AND connect_no  EQ @pa_con
     AND flight_date EQ @pa_fld
     AND airplane_type EQ @pa_pla.

  IF sy-subrc NE 0.
    MESSAGE s001.
    LEAVE LIST-PROCESSING.
  ENDIF.

ENDFORM.

FORM set_fcat_layout .

  gs_layout-zebra      = 'X'.
  gs_layout-cwidth_opt = 'A'.
  gs_layout-sel_mode   = 'D'.

  gs_stable-row = 'X'.
  gs_stable-col = 'X'.

  CALL FUNCTION 'LVC_FIELDCATALOG_MERGE'
    EXPORTING
      i_structure_name = 'ZC505CDS0005'
    CHANGING
      ct_fieldcat      = gt_fcat.

  DELETE gt_fcat INDEX 1.

  LOOP AT gt_fcat ASSIGNING FIELD-SYMBOL(<fs_fcat>).

    CASE <fs_fcat>-fieldname.

      WHEN 'CARRIER' OR 'CONNECT_NO' OR 'FLIGHT_DATE' OR 'BOOKING_ID'.
        <fs_fcat>-key = 'X'.

      WHEN 'AIRPLANE_TYPE'.
        <fs_fcat>-coltext = 'AIRPLANE_TYPE'.

*      WHEN 'PAYMENTSUM'.
*        <fs_fcat>-cfieldname = 'CURRENCY'.

    ENDCASE.

  ENDLOOP.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left " Side to Which Control is Docked
      extension = 5000.                       " Control Extension

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant       " Layout
      i_save          = 'A'              " Save Layout
      i_default       = 'X'              " Default Display Variant
      is_layout       = gs_layout        " Layout
    CHANGING
      it_outtab       = gt_data           " Output Table
      it_fieldcatalog = gt_fcat.         " Field Catalog

ENDFORM.
