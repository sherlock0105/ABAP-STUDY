FORM get_aufk .

  REFRESH gt_aufk.

  DATA : lr_aufnr TYPE RANGE OF aufk-aufnr.

  IF pa_auf IS NOT INITIAL.

    lr_aufnr = VALUE #( BASE lr_aufnr

                      ( low    = pa_auf
                        sign   = 'I'
                        option = 'EQ' )

                       ).

  ENDIF.

  SELECT *
    INTO TABLE @gt_aufk
    FROM zc505cds0011
   WHERE aufnr IN @lr_aufnr.

ENDFORM.

FORM display_screen .

  CREATE OBJECT gcl_container
    EXPORTING
      side      = gcl_container->dock_at_left " Side to Which Control is Docked
      extension = 5000.                       " Control Extension

  CREATE OBJECT gcl_grid
    EXPORTING
      i_parent = gcl_container.

  gs_variant-report = sy-repid.

  CALL METHOD gcl_grid->set_table_for_first_display
    EXPORTING
      is_variant      = gs_variant       " Layout
      i_save          = 'A'              " Save Layout
      i_default       = 'X'              " Default Display Variant
      is_layout       = gs_layout        " Layout
    CHANGING
      it_outtab       = <gft_itab>          " Output Table
      it_fieldcatalog = gt_fcat.         " Field Catalog

  UNASSIGN <gft_itab>.

ENDFORM.

FORM control_screen .

  LOOP AT SCREEN.

    IF pa_rb1 = 'X'.

      IF screen-group1 = 'RB2'.
        screen-active = 0.
      ENDIF.

    ENDIF.

    IF pa_rb2 = 'X'.

      IF screen-group1 = 'RB1'.
        screen-active = 0.
      ENDIF.

    ENDIF.

    MODIFY SCREEN.

  ENDLOOP.

ENDFORM.
